\begin{frame}{Módulo 2: ¿qué es una red VPC?}
\begin{block}{Definición}
\small La \textbf{red privada de tu proyecto} (\emph{Virtual Private Cloud}). Es donde se conectan tus VMs, como el wifi de tu casa conecta tus equipos.
\end{block}
\begin{columns}[c,onlytextwidth]
\column{.52\textwidth}
\centering
\begin{tikzpicture}[font=\footnotesize]
  \draw[utpRojo,thick,rounded corners=6pt,fill=utpRojo!5] (0,0) rectangle (6.4,3.3);
  \node[anchor=north west,font=\small\bfseries] at (0.1,3.25) {Red VPC: el condominio};
  \draw[rounded corners=4pt,fill=white] (0.3,0.3) rectangle (3.0,2.6);
  \node[anchor=north west] at (0.35,2.55) {\textbf{Subred en EE.\,UU.}};
  \node[anchor=north west] at (0.35,2.15) {un bloque};
  \node[draw,fill=utpRojo!20,rounded corners=2pt] at (1.0,0.9) {VM a};
  \node[draw,fill=utpRojo!20,rounded corners=2pt] at (2.3,0.9) {VM b};
  \draw[rounded corners=4pt,fill=white] (3.4,0.3) rectangle (6.1,2.6);
  \node[anchor=north west] at (3.45,2.55) {\textbf{Subred en Chile}};
  \node[anchor=north west] at (3.45,2.15) {otro bloque};
  \node[draw,fill=utpRojo!20,rounded corners=2pt] at (4.75,0.9) {VM c};
\end{tikzpicture}
\column{.44\textwidth}
\begin{exampleblock}{Una comparación}
\small La VPC es un \textbf{condominio}; las \textbf{subredes} son los bloques de edificios; las \textbf{VMs}, los departamentos.
\end{exampleblock}
\begin{block}{Lo sorprendente}
\small Cada subred está en \textbf{una región}, pero la red las une a todas: la VM de Chile y la de EE.\,UU. se hablan como vecinas, sin salir a Internet.
\end{block}
\end{columns}
\par\vspace{1mm}
\begin{alertblock}{Ya la usaste}
\small Cada proyecto trae una red llamada \texttt{default}. Tus VMs de la Sesión 14 vivían ahí.
\end{alertblock}
\end{frame}

\begin{frame}{Módulo 2: subredes y sus direcciones}
\begin{block}{Cada subred tiene un rango de direcciones}
\small Se escribe así: \texttt{10.0.1.0/24}. El número después de la barra dice el \textbf{tamaño}: mientras \textbf{más chico}, \textbf{más direcciones}.
\end{block}
\renewcommand{\arraystretch}{1.3}
\centering\small
\begin{tabular}{@{}p{2.4cm}p{3.4cm}p{3.0cm}p{3.8cm}@{}}
\toprule
\textbf{Rango} & \textbf{Direcciones} & \textbf{Para VMs} & \textbf{Alcanza para\ldots}\\
\midrule
\texttt{/29} & 8 & 4 & una prueba pequeña\\
\texttt{/24} & 256 & 252 & un proyecto normal\\
\texttt{/20} & 4\,096 & 4\,092 & una empresa grande\\
\bottomrule
\end{tabular}
\par\vspace{3mm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{¿Por qué 4 menos?}
\small Google se \textbf{reserva 4 direcciones} en cada subred para que la red funcione (la primera, la puerta de salida y las dos últimas).
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{Si te quedas corto}
\small Puedes \textbf{agrandar} la subred sin apagar nada. Lo que no se puede es achicarla.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Módulo 2: IP interna e IP externa}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{IP interna}
\small La tiene \textbf{toda} VM. Sirve para hablar \textbf{dentro} de la red, con las otras VMs.
\end{block}
\begin{block}{IP externa}
\small \textbf{Opcional}. Es la que abriste desde el navegador en el lab de NGINX.\par\smallskip
\textbf{Efímera:} puede cambiar si apagas la VM.\par
\textbf{Estática:} la reservas y se queda fija.
\end{block}
\column{.48\textwidth}
\begin{exampleblock}{Una comparación}
\small La IP interna es el \textbf{anexo} de una oficina: solo funciona dentro. La externa es el \textbf{número público} de la central: cualquiera puede llamar.
\end{exampleblock}
\begin{alertblock}{Saberes previos, resuelto}
\small El \texttt{192.168.1.25} de tu celular es como una IP \textbf{interna}: desde Internet no se ve. Tu casa sale con la IP pública del router.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Módulo 2: el firewall, el portero de la red}
\begin{center}
\begin{tikzpicture}[font=\small,
  caja/.style={draw=utpRojo,rounded corners=4pt,minimum width=2.4cm,minimum height=1.1cm,align=center}]
  \node[caja,fill=white] (i) at (0,0) {\textbf{Internet}};
  \node[caja,fill=utpRojo!30] (f) at (5.2,0) {\textbf{Firewall}\\\footnotesize revisa su lista};
  \node[caja,fill=utpRojo!10] (v) at (10.4,0) {\textbf{Tu VM}};
  \draw[-latex,thick] (i) -- node[above,font=\footnotesize] {visita al puerto 80} (f);
  \draw[-latex,thick] (f) -- node[above,font=\footnotesize] {¿está en la lista?} (v);
\end{tikzpicture}
\end{center}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{La regla básica}
\small Lo que \textbf{entra}: todo \textbf{bloqueado}, salvo lo que tú permitas.\par\smallskip
Lo que \textbf{sale}: todo permitido.
\end{block}
\begin{exampleblock}{Una regla dice}
\small \textbf{Qué} (puerto 80, web), \textbf{desde dónde} (cualquier IP) y \textbf{a quién} (las VMs con cierta etiqueta).
\end{exampleblock}
\column{.48\textwidth}
\begin{block}{Los puertos más comunes}
\small
\textbf{22:} SSH, entrar a la terminal.\par
\textbf{80:} web (\texttt{http://}).\par
\textbf{443:} web segura (\texttt{https://}).
\end{block}
\begin{alertblock}{Una comparación}
\small El portero del condominio con su \textbf{lista de invitados}: si no estás, no entras.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Módulo 2: la casilla ``Permitir tráfico HTTP''}
\begin{block}{Qué hizo en la Sesión 14}
\small Le puso a la VM una \textbf{etiqueta}: \texttt{http-server}. La red ya tenía una regla: ``a las VMs con esa etiqueta, deja pasar la web (puerto 80)''.
\end{block}
\begin{center}
\begin{tikzpicture}[font=\small,
  caja/.style={draw=utpRojo,rounded corners=4pt,minimum width=3.6cm,minimum height=1.1cm,align=center}]
  \node[caja,fill=utpRojo!10] (c) at (0,0) {Marcas la casilla};
  \node[caja,fill=utpRojo!20] (e) at (4.6,0) {VM con etiqueta\\\texttt{http-server}};
  \node[caja,fill=utpRojo!35] (r) at (9.2,0) {La regla la deja\\pasar: la web abre};
  \draw[-latex,thick] (c) -- (e);
  \draw[-latex,thick] (e) -- (r);
\end{tikzpicture}
\end{center}
\begin{exampleblock}{La misma regla, escrita como comando}
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize]
gcloud compute firewall-rules create permitir-web \
  --allow=tcp:80 --target-tags=http-server
\end{lstlisting}
\end{exampleblock}
\begin{alertblock}{Error típico}
\small Crear la regla pero olvidar ponerle la \textbf{etiqueta} a la VM: la página no abre.
\end{alertblock}
\end{frame}

\begin{frame}{Módulo 2: VMs sin IP externa y lo que cuesta la red}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{VMs escondidas, pero útiles}
\small Una VM \textbf{sin IP externa} es más segura: nadie la alcanza desde Internet. Pero, ¿cómo descarga actualizaciones?\par\smallskip
\textbf{Cloud NAT} le da \textbf{salida} a Internet, sin dejar que nadie entre.
\end{block}
\begin{exampleblock}{Una comparación}
\small Como una casa con \textbf{puerta solo de salida}: tú sales a comprar, pero nadie toca el timbre.
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{Qué se cobra}
\small
Datos que \textbf{entran}: gratis.\par\smallskip
Datos entre VMs \textbf{cercanas} (misma zona): gratis.\par\smallskip
Datos que \textbf{salen a Internet}: se cobran.
\end{alertblock}
\begin{block}{Idea clave}
\small Lo que \textbf{sale} cuesta. Mantener las cosas cerca ahorra dinero.
\end{block}
\end{columns}
\end{frame}

% ---- Modulo 3
\begin{frame}{Módulo 3: una VM por dentro}
\begin{block}{Qué eliges al crear una VM}
\small Lo mismo que al comprar una laptop: \textbf{procesador} (vCPU), \textbf{memoria}, \textbf{disco} y \textbf{sistema operativo}. Más \textbf{dónde vive} (zona) y \textbf{a qué red} se conecta.
\end{block}
\renewcommand{\arraystretch}{1.3}
\centering\small
\begin{tabular}{@{}p{3.6cm}p{2.8cm}p{6.2cm}@{}}
\toprule
\textbf{Familia} & \textbf{Ejemplo} & \textbf{Como una\ldots}\\
\midrule
Uso general & E2, N2 & \textbf{laptop de oficina}: sirve para casi todo\\
Mucho cálculo & C2 & \textbf{PC gamer}: procesador potente\\
Mucha memoria & M3 & \textbf{servidor gigante} para bases de datos enormes\\
Con GPU & A3, G2 & \textbf{equipo de IA}: tarjetas gráficas\\
\bottomrule
\end{tabular}
\par\vspace{3mm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Cómo se lee \texttt{e2-medium}}
\small Familia \textbf{E2}, tamaño \textbf{medium}. Es la que usamos en los labs.
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{Para empezar}
\small Casi siempre: \textbf{E2}. Es la más barata.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Módulo 3: discos, imágenes y copias}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\begin{block}{Imagen}
\small El \textbf{sistema operativo} con el que nace la VM: Debian, Ubuntu, Windows\ldots{}\par\smallskip
También puedes crear tu \textbf{propia imagen} de una VM ya configurada.
\end{block}
\column{.31\textwidth}
\begin{block}{Disco}
\small Donde se guardan los archivos y el sistema operativo.\par\smallskip
Se puede \textbf{agrandar}, no achicar.
\end{block}
\column{.31\textwidth}
\begin{block}{Instantánea}
\small Una \textbf{foto} del disco en un momento. Sirve como copia de seguridad o para clonar la VM.
\end{block}
\end{columns}
\par\vspace{3mm}
\begin{exampleblock}{Una comparación}
\small La \textbf{imagen} es el molde de la torta; el \textbf{disco}, la torta que te comes; la \textbf{instantánea}, la foto que le tomas antes de cortarla.
\end{exampleblock}
\begin{alertblock}{Script de inicio}
\small Un pequeño programa que la VM corre \textbf{cada vez que arranca}. Así se puede instalar NGINX sin entrar por SSH.
\end{alertblock}
\end{frame}
