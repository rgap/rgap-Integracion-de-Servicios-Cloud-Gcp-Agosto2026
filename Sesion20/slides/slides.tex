\separador{Parte 1: las redes VPC, en corto}{Qué es una red VPC y qué es una ruta}

\begin{frame}{¿Qué es una red VPC?}
\begin{block}{En una frase}
\small La \textbf{red privada} de tu proyecto dentro de Google Cloud: ahí se conectan tus máquinas virtuales. Se arma con clics o comandos, \textbf{sin cables}.
\end{block}
\begin{columns}[c,onlytextwidth]
\column{.54\textwidth}
\centering
\begin{tikzpicture}[font=\footnotesize]
  \draw[black!55,rounded corners=6pt] (-0.25,-0.25) rectangle (6.85,4.2);
  \node[anchor=north west,font=\small\bfseries] at (-0.15,4.15) {Proyecto};
  \draw[utpRojo,thick,rounded corners=6pt,fill=utpRojo!5] (0,0) rectangle (6.6,3.5);
  \node[anchor=north west,font=\small\bfseries] at (0.1,3.45) {Red (es mundial)};
  \draw[rounded corners=4pt,fill=white] (0.3,0.3) rectangle (3.1,2.7);
  \node[anchor=north west] at (0.35,2.65) {\textbf{Subred}};
  \node[anchor=north west] at (0.35,2.25) {región de EE.\,UU.};
  \node[draw,fill=utpRojo!20,rounded corners=2pt] at (1.05,0.9) {VM};
  \node[draw,fill=utpRojo!20,rounded corners=2pt] at (2.3,0.9) {VM};
  \draw[rounded corners=4pt,fill=white] (3.5,0.3) rectangle (6.3,2.7);
  \node[anchor=north west] at (3.55,2.65) {\textbf{Subred}};
  \node[anchor=north west] at (3.55,2.25) {región de Europa};
  \node[draw,fill=utpRojo!20,rounded corners=2pt] at (4.9,0.9) {VM};
\end{tikzpicture}
\column{.43\textwidth}
\begin{block}{De afuera hacia adentro}
\small
\textbf{Proyecto:} la caja donde vive todo.\par\smallskip
\textbf{Red:} abarca \textbf{todo el mundo}.\par\smallskip
\textbf{Subred:} un pedazo de la red en \textbf{una región}.\par\smallskip
\textbf{VM:} vive dentro de una subred.
\end{block}
\begin{exampleblock}{Una comparación}
\small Una \textbf{empresa} con una \textbf{sede} en cada ciudad y \textbf{empleados} en cada sede.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Las redes VPC en cinco ideas}
\renewcommand{\arraystretch}{1.4}
\centering\small
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{3.7cm}>{\raggedright\arraybackslash}p{5.3cm}>{\raggedright\arraybackslash}p{3.9cm}@{}}
\toprule
\textbf{La idea} & \textbf{Qué significa} & \textbf{Se parece a}\\
\midrule
\textbf{La red es mundial} & Una sola red llega a todas las regiones & Una empresa con sedes en varios países\\
\textbf{La subred es de una región} & Cada una tiene su propio grupo de direcciones & La sede de una ciudad\\
\textbf{Misma red, se hablan} & Por dentro, aunque estén lejos. Entre redes distintas, no & Los anexos de una misma empresa\\
\textbf{Dos direcciones} & La IP interna la tienen todas. La externa es opcional & El anexo y el número público\\
\textbf{El firewall decide} & Lo que entra está bloqueado hasta que una regla lo permita & El portero con su lista\\
\bottomrule
\end{tabular}
\par\vspace{2mm}
\begin{alertblock}{La idea que abre la parte 2}
\small La IP externa es \textbf{opcional}. ¿Qué pasa con una máquina que \textbf{no la tiene}?
\end{alertblock}
\end{frame}

\begin{frame}{¿Qué es una ruta?}
\begin{center}
{\Large Una ruta es un \textbf{letrero}: ``para ir \textbf{allá}, ve por \textbf{aquí}''.}
\end{center}
\vspace{1mm}
\begin{center}
\begin{tikzpicture}[font=\small,
  caja/.style={draw=utpRojo,rounded corners=4pt,minimum width=3.2cm,minimum height=0.9cm,align=center}]
  \node[caja,fill=white] (v) at (0,0) {\textbf{Tu VM}};
  \node[caja,fill=utpRojo!10] (a) at (8,0.9) {\textbf{Otra VM de tu red}};
  \node[caja,fill=utpRojo!30] (b) at (8,-0.9) {\textbf{Internet}};
  \draw[-latex,thick] (v.east) -- node[above,sloped,font=\footnotesize] {por dentro} (a.west);
  \draw[-latex,thick] (v.east) -- node[below,sloped,font=\footnotesize] {hacia afuera} (b.west);
\end{tikzpicture}
\end{center}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Como en la carretera}
\small ``Arequipa $\rightarrow$''. El letrero no maneja por ti: solo te indica el camino.
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{Pero el letrero no basta}
\small Para salir a Internet, la VM necesita una \textbf{IP externa}. ¿Y si no tiene? Eso es la parte 2.
\end{alertblock}
\end{columns}
\par\vspace{2mm}
\centering\small Google pone estos dos letreros en toda red. \textbf{Tú no haces nada.}
\end{frame}

\begin{center}

\begin{columns}[T,onlytextwidth]

\begin{block}{Qué es}
\small Un \textbf{interruptor} de la \textbf{subred}. Lo enciendes y sus VMs llegan a los servicios de Google, como Cloud Storage.
\end{block}
\begin{alertblock}{Solo llega a Google}
\small Las actualizaciones de Internet \textbf{siguen sin llegar}.
\end{alertblock}
\column{.48\textwidth}
\begin{exampleblock}{Como en una empresa}
\small Un \textbf{pasillo interno} hasta el almacén: llegas sin salir a la calle.
\end{exampleblock}
\begin{block}{Dónde se enciende}
\small En la \textbf{subred}, no en la VM. Vale para todas las VMs de esa subred.
\end{block}

\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Qué es}
\small Deja a la VM \textbf{salir} a Internet. La respuesta vuelve, pero nadie de afuera puede \textbf{empezar} una conexión hacia ella.
\end{block}
\column{.48\textwidth}
\begin{exampleblock}{Ya lo usas en casa}
\small Tu \textbf{router wifi}: tu celular navega y nadie de Internet entra a él.
\end{exampleblock}
\end{columns}
\par\vspace{1mm}
\begin{block}{¿Y NAT?}
\small \emph{Traducción de direcciones}: al salir, cambia la IP interna de la VM por una pública.
\end{block}
\end{frame}

\end{alertblock}

\begin{frame}{El lab: tres ideas para no perderte}
\renewcommand{\arraystretch}{1.45}
\centering\small
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{4.3cm}>{\raggedright\arraybackslash}p{4.6cm}>{\raggedright\arraybackslash}p{4.0cm}@{}}
\toprule
\textbf{La idea} & \textbf{Dónde la ves en el lab} & \textbf{Si la olvidas}\\
\midrule
\textbf{La VM nace sin IP externa} & Al crearla, en IP externa eliges \textbf{Ninguna} & El lab pierde sentido: todo funciona desde el inicio\\
\textbf{El Acceso privado es de la subred} & Editas la \textbf{subred}, no la máquina & Lo buscas en la VM y no aparece\\
\textbf{Cloud NAT no es instantáneo} & Tarda hasta \textbf{3 minutos} en hacer efecto & Crees que falló: espera y repite\\
\bottomrule
\end{tabular}
\par\vspace{3mm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{¿Y cómo entras a la VM?}
\small No tiene IP externa, así que Google te abre un \textbf{túnel seguro} (IAP). El comando ya viene escrito en el lab.
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{Si algo falla}
\small Fíjate \textbf{dónde estás}: Cloud Shell y la VM son dos lugares distintos. Lo que guardaste en uno no existe en el otro.
\end{alertblock}
\end{columns}
\end{frame}
