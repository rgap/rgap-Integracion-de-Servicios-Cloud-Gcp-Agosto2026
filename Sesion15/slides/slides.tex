\begin{frame}{Repaso: WSL y virtualización anidada}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{WSL}
\small \textbf{Linux real dentro de Windows}, sin reiniciar ni instalar otro sistema. Se instala con \texttt{wsl -{}-install}.\par\smallskip
\textbf{WSL 2} usa un kernel de Linux verdadero dentro de una VM ligera.
\end{block}
\begin{exampleblock}{Para qué sirve de verdad}
\small Trabajar con las mismas herramientas que el servidor: Python, Node.js, Docker Desktop, \texttt{gcloud}, \texttt{ssh}, \texttt{git}.
\end{exampleblock}
\column{.48\textwidth}
\begin{block}{Virtualización anidada}
\small Una \textbf{VM dentro de otra VM}, como las matrioskas. La VM de afuera debe recibir las funciones del procesador (VT-x o AMD-V).
\end{block}
\begin{alertblock}{La conexión entre las dos}
\small WSL 2 \textbf{ya es una VM}. Si tu Windows es una VM, usar WSL 2 ahí es virtualización anidada.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Repaso: sitios estáticos}
\begin{block}{Idea clave}
\small Un sitio estático es solo \textbf{archivos} (HTML, CSS, JS). El servidor no calcula nada: los entrega tal como están. Por eso se publica gratis.
\end{block}
\renewcommand{\arraystretch}{1.3}
\centering\small
\begin{tabular}{@{}p{3.4cm}p{4.6cm}p{4.6cm}@{}}
\toprule
& \textbf{GitHub Pages} & \textbf{Vercel}\\
\midrule
Cómo se publica & Settings $\rightarrow$ Pages $\rightarrow$ rama \texttt{main} & Importar repo, preset \textbf{Other}, Deploy\\
Dirección & \texttt{usuario.github.io/repo} & \texttt{proyecto.vercel.app}\\
Vista previa por rama & No & Sí\\
\bottomrule
\end{tabular}
\par\vspace{3mm}
\begin{alertblock}{Los dos errores más comunes}
\small Rutas con \texttt{/} al inicio (\texttt{/styles.css}) y nombres con otra mayúscula (\texttt{Styles.css}): la página sale sin estilos.
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{Repaso: servicios y tu primera VM}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Habilitar servicios}
\small En un proyecto nuevo, las APIs están \textbf{apagadas}. Se encienden en \textbf{APIs y servicios} $\rightarrow$ \textbf{Biblioteca}, o con:
\begin{lstlisting}
gcloud services enable \
  compute.googleapis.com
\end{lstlisting}
\end{block}
\begin{alertblock}{IAM + servicio}
\small IAM decide \textbf{quién} lo usa; habilitar decide \textbf{si existe} en tu proyecto. Necesitas las dos.
\end{alertblock}
\column{.48\textwidth}
\begin{exampleblock}{El laboratorio de la VM}
\small
\textbf{1.} VM \texttt{gcelab} desde la consola con \emph{Permitir tráfico HTTP}.\par\smallskip
\textbf{2.} SSH, instalar \textbf{NGINX} y abrir la IP externa con \texttt{http://}.\par\smallskip
\textbf{3.} VM \texttt{gcelab2} con \texttt{gcloud compute instances create}.
\end{exampleblock}
\begin{block}{Lo que era}
\small Compute Engine es \textbf{IaaS}: te dan la máquina y tú instalas y cuidas todo lo de adentro.
\end{block}
\end{columns}
\end{frame}

\begin{frame}{¿Qué es una certificación de Google Cloud?}
\begin{block}{Definición}
\small Un \textbf{examen oficial} que demuestra que sabes hacer algo en Google Cloud. Un curso te enseña; la certificación \textbf{comprueba} que lo aprendiste.
\end{block}

\begin{exampleblock}{Una comparación}
\small Como el \textbf{brevete}: puedes saber manejar, pero el brevete es la prueba oficial que te pide el empleador.
\end{exampleblock}
\begin{alertblock}{Hoy nos enfocamos en}
\small \textbf{Associate Cloud Engineer (ACE)}: el primer nivel técnico, pensado para quien crea y administra cosas en Google Cloud.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{¿Qué hace un Cloud Engineer?}
\begin{block}{En una frase}
\small \textbf{Crea, configura y cuida} lo que una empresa tiene en la nube: máquinas, bases de datos, redes, permisos y costos.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Una comparación}
\small Es el \textbf{administrador de un edificio}: no diseñó el edificio (eso es el arquitecto), pero reparte las llaves, revisa que funcione la luz, atiende fallas y controla los gastos.
\end{exampleblock}
\column{.48\textwidth}
\begin{block}{Un día normal}
\small
Crear una VM o desplegar en Cloud Run.\par\smallskip
Dar acceso a un nuevo compañero (IAM).\par\smallskip
Revisar por qué una app se cayó (registros).\par\smallskip
Poner una alerta si el gasto sube.
\end{block}
\end{columns}
\par\vspace{1mm}
\begin{alertblock}{Ya lo empezaste}
\small Crear un proyecto, habilitar una API, dar un rol y crear una VM: eso es trabajo de Cloud Engineer.
\end{alertblock}
\end{frame}

\begin{frame}{El examen Associate Cloud Engineer}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Cómo es}
\small
\textbf{Duración:} 2 horas.\par\smallskip
\textbf{Preguntas:} entre 50 y 60, de opción múltiple (una o varias respuestas).\par\smallskip
\textbf{Dónde:} en línea desde tu casa (con supervisión) o en un centro de evaluación.\par\smallskip
\textbf{Costo:} alrededor de 125 dólares.\par\smallskip
\textbf{Vigencia:} 3 años.
\end{block}
\column{.48\textwidth}
\begin{exampleblock}{Cómo son las preguntas}
\small Casi siempre son \textbf{situaciones}: ``Tu equipo necesita\ldots{} ¿qué haces?''. No piden memorizar, piden \textbf{elegir la mejor opción}.
\end{exampleblock}
\begin{block}{Experiencia sugerida}
\small Google recomienda unos \textbf{6 meses} de práctica con Google Cloud. Los labs cuentan como práctica.
\end{block}
\end{columns}
\par\vspace{1mm}
\begin{alertblock}{Antes de inscribirte}
\small Los datos del examen cambian: revisa la página oficial en \texttt{cloud.google.com/learn/certification/cloud-engineer}.
\end{alertblock}
\end{frame}

\begin{frame}{Regiones y zonas: dónde vive tu recurso}
\begin{block}{Definiciones}
\small Una \textbf{región} es un lugar del mundo con centros de datos de Google (por ejemplo, \texttt{southamerica-west1} en Santiago de Chile). Cada región tiene varias \textbf{zonas}: edificios separados dentro de ella (\texttt{southamerica-west1-a}, \texttt{-b}, \texttt{-c}).
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Una comparación}
\small La región es la \textbf{ciudad}; las zonas son \textbf{distritos} con su propia luz. Si se corta en uno, los otros siguen.
\end{exampleblock}
\begin{block}{Recursos zonales y regionales}
\small Una VM vive en \textbf{una zona}. Otros servicios, como Cloud Run o un bucket regional, viven en \textbf{una región}.
\end{block}
\column{.48\textwidth}
\begin{block}{Cómo elegir}
\small
\textbf{Cerca de tus usuarios:} responde más rápido.\par\smallskip
\textbf{Precio:} cambia según la región.\par\smallskip
\textbf{Plan gratuito:} algunas ofertas gratis solo aplican en ciertas regiones de EE.\,UU.
\end{block}
\begin{alertblock}{Error típico}
\small ``\texttt{gcloud} me pide zona'': falta \texttt{-{}-zone}. Recuerda el lab de la VM.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Tres formas de trabajar en Google Cloud}
\begin{columns}[T,onlytextwidth]
\column{.30\textwidth}
\begin{block}{Consola}
\small La página web \texttt{console.cloud.google.com}. Con clics. Ideal para empezar y para \textbf{ver} lo que existe.
\end{block}
\column{.30\textwidth}
\begin{block}{Cloud Shell}
\small Una terminal Linux \textbf{en el navegador}, con \texttt{gcloud} ya instalado. Gratis y sin instalar nada.
\end{block}
\column{.34\textwidth}
\begin{block}{\texttt{gcloud}}
\small La herramienta de \textbf{línea de comandos}. Se puede instalar también en tu laptop o en WSL.
\end{block}
\end{columns}
\par\vspace{1mm}
\begin{exampleblock}{Primeros comandos que verás en la ruta}
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize]
gcloud auth list                                  # con que cuenta estoy
gcloud config list project                        # en que proyecto estoy
gcloud config set compute/region us-central1      # region por defecto
gcloud compute instances list                     # mis VMs
\end{lstlisting}
\end{exampleblock}
\begin{alertblock}{Consejo}
\small El examen pregunta por los dos caminos. Si lo haces con clics, busca también el comando.
\end{alertblock}
\end{frame}

\begin{frame}{Cómputo: ¿VM, Cloud Run o GKE?}
\begin{block}{Una comparación para moverse por la ciudad}
\small Las tres llevan tu aplicación a Internet; cambia \textbf{cuánto manejas tú}.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\begin{exampleblock}{Compute Engine}
\small \textbf{Tu propio auto.} Eliges el modelo, pones la gasolina y lo llevas al taller. Control total, más trabajo.
\end{exampleblock}
\column{.31\textwidth}
\begin{exampleblock}{Cloud Run}
\small \textbf{Un taxi.} Dices a dónde vas y pagas solo el viaje. Si nadie usa tu app, casi no pagas.
\end{exampleblock}
\column{.31\textwidth}
\begin{exampleblock}{GKE}
\small \textbf{Una flota de buses.} Muchos vehículos coordinados. Potente, pero hay que aprender a administrarla.
\end{exampleblock}
\end{columns}
\par\vspace{2mm}
\begin{alertblock}{Pregunta típica del examen}
\small ``Tienes un contenedor, el tráfico es irregular y quieres pagar poco cuando no hay visitas''. Respuesta: \textbf{Cloud Run}.
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{Almacenamiento: Cloud Storage}
\begin{block}{Qué es}
\small Un lugar para guardar \textbf{archivos} (imágenes, PDF, copias de seguridad) dentro de \textbf{buckets}: carpetas con un nombre único en todo el mundo.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Clases de almacenamiento}
\small
\textbf{Standard:} lo que se usa todo el tiempo.\par\smallskip
\textbf{Nearline} y \textbf{Coldline:} lo que se usa poco.\par\smallskip
\textbf{Archive:} lo que casi nunca se abre. Guardar es muy barato; leer cuesta más.
\end{exampleblock}
\column{.48\textwidth}
\begin{block}{Con \texttt{gcloud}}
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize]
gcloud storage buckets create \
  gs://NOMBRE-UNICO \
  --location=us-central1

gcloud storage cp foto.png \
  gs://NOMBRE-UNICO/
\end{lstlisting}
\end{block}
\end{columns}
\par\vspace{1mm}
\begin{alertblock}{Relación con la Sesión 14}
\small Un bucket también puede servir una página estática, como GitHub Pages, pero dentro de Google Cloud.
\end{alertblock}
\end{frame}

\begin{frame}{Costos: el tema que nadie quiere olvidar}
\begin{block}{Idea clave}
\small En la nube pagas por lo que \textbf{usas}, y lo que olvidas encendido \textbf{también se usa}. El Cloud Engineer vigila el gasto.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Para aprender sin gastar}
\small
\textbf{Prueba gratuita:} crédito para cuentas nuevas (hoy, 300 dólares por 90 días).\par\smallskip
\textbf{Nivel gratuito:} algunos servicios tienen un uso mensual gratis, como una VM pequeña \texttt{e2-micro}.\par\smallskip
\textbf{Labs de Google Skills:} usan un proyecto temporal; no se cobra a tu tarjeta.
\end{exampleblock}
\column{.48\textwidth}
\begin{block}{Presupuestos y alertas}
\small En \textbf{Facturación} $\rightarrow$ \textbf{Presupuestos y alertas} defines un monto (por ejemplo, 10 dólares) y Google te avisa por correo al 50\,\%, 90\,\% y 100\,\%.
