\begin{frame}{Visual Studio Code edita el proyecto en Ubuntu}
\begin{block}{Para qué sirve}
\small VS Code es un editor gráfico. Permite abrir la carpeta del curso, editar los scripts y usar una terminal integrada sin salir del editor.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.52\textwidth}
\begin{exampleblock}{1. Instala el paquete \texttt{.deb}}
\begin{terminal}\footnotesize
\$ dpkg --print-architecture\\
arm64\\
\$ cd \textasciitilde/Descargas\\
\$ sudo apt install ./code\_*.deb
\end{terminal}
\small Descarga primero el paquete para Debian/Ubuntu de tu arquitectura desde \texttt{code.visualstudio.com/download}.
\end{exampleblock}
\column{.44\textwidth}
\begin{exampleblock}{2. Abre la carpeta del curso}
\begin{terminal}\footnotesize
\$ code \textasciitilde/Escritorio/curso\_cloud
\end{terminal}
\small También puedes abrirlo desde \textbf{Actividades} buscando \texttt{Visual Studio Code}.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Warp añade ayuda de IA sin cambiar tu shell}
\begin{block}{Para qué sirve}
\small Warp es una terminal gráfica que puede explicar comandos, sugerir pasos y ayudar a trabajar con proyectos. En Ubuntu puede seguir utilizando Bash.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.43\textwidth}
\begin{exampleblock}{1. Comprueba la arquitectura}
\begin{terminal}
\$ dpkg --print-architecture\\
arm64
\end{terminal}
\small La salida también puede ser \texttt{amd64}. Descarga el paquete \texttt{.deb} correspondiente.
\end{exampleblock}
\column{.53\textwidth}
\begin{exampleblock}{2. Instala y abre Warp}
\begin{terminal}\footnotesize
\$ cd \textasciitilde/Descargas\\
\$ sudo apt install ./warp-terminal\_*.deb\\
\$ warp-terminal
\end{terminal}
\small Descarga primero el paquete para Debian/Ubuntu desde \texttt{warp.dev/download}.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Antigravity CLI ayuda dentro del proyecto mediante \texttt{agy}}
\begin{block}{Para qué sirve}
\small \texttt{agy} abre un agente de IA en la terminal. Puede revisar archivos del proyecto, explicar código y proponer cambios sujetos a tu confirmación.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.57\textwidth}
\begin{exampleblock}{Instalación oficial en Ubuntu}
\begin{terminal}\footnotesize
\$ sudo apt update\\
\$ sudo apt install curl -y\\
\$ curl -fsSL \textbackslash\\
\quad https://antigravity.google/cli/install.sh | bash\\
\$ source \textasciitilde/.bashrc
\end{terminal}
\end{exampleblock}
\column{.39\textwidth}
\begin{exampleblock}{Ejemplo en el laboratorio}
\begin{terminal}\footnotesize
\$ cd \textasciitilde/Escritorio/curso\_cloud/sesion06\\
\$ agy
\end{terminal}
\small Solicitud de ejemplo:\par
\texttt{Explica qué interfaces de red tiene esta VM.}
\end{exampleblock}
\end{columns}
\begin{alertblock}{Primer inicio y ruta del comando}
\scriptsize El primer inicio puede abrir el navegador para iniciar sesión. Si aparece \texttt{agy: command not found}, agrega \texttt{export PATH="\$HOME/.local/bin:\$PATH"} a \texttt{\textasciitilde/.bashrc} y ejecuta \texttt{source \textasciitilde/.bashrc}.
\end{alertblock}
\end{frame}

\begin{frame}{Práctica: reúne los datos de la ruta \textnormal{\small(12 min)}}
\begin{block}{Reto}
\small Los comandos no se entregan: descúbrelos tú. Puedes apoyarte en Warp o en \texttt{agy} para encontrarlos, pero anota cuál usaste para cada dato.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Direcciones de origen y salida}
\small
\texttt{VM\_IP}: la IPv4 activa de la VM.\par\medskip
\texttt{ROUTER\_IP}: la puerta de enlace por la que sale el tráfico de la VM.
\end{exampleblock}
\column{.48\textwidth}
\begin{exampleblock}{Internet y destino}
\small
\texttt{PUBLIC\_IP}: la IP con la que Internet ve tu salida.\par\medskip
\texttt{DEST\_IP}: una IPv4 de Google; usa \texttt{Google} como \texttt{DEST\_NAME}.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}[t]{Observa el tráfico al abrir Google en Firefox}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{exampleblock}{En la Mac: \texttt{HOST\_IP}}
\small Averigua la IPv4 que la Mac tiene en la red Wi-Fi y regístrala como \texttt{HOST\_IP}. Primero necesitas saber qué interfaz corresponde al Wi-Fi.
\end{exampleblock}
\begin{block}{Modo de red}
\small Confirma en la configuración del hipervisor: \textbf{Red = Bridged}. Registra \texttt{MODE=Bridged}.
\end{block}
\column{.49\textwidth}
\begin{exampleblock}{En la VM: protocolo y puerto}
\small Captura el tráfico hacia \texttt{DEST\_IP} mientras abres \texttt{google.com} en Firefox y detén la captura cuando cargue la página. A partir de lo capturado deduce y registra \texttt{PROTOCOL} y \texttt{PORT}.
\end{exampleblock}
\end{columns}
\begin{alertblock}{Evidencia obligatoria}
\scriptsize Anota junto a cada valor el comando utilizado. El modo puente se sustenta con la configuración de la VM, no con un comando dentro de Ubuntu.
\end{alertblock}
\end{frame}

\begin{frame}{Convierte los hallazgos en un prompt de visualización}
\begin{columns}[T,onlytextwidth]
\column{.40\textwidth}
\begin{exampleblock}{Completa la ficha}
\begin{terminal}\scriptsize
VM\_IP     = [VM\_IP]\\
HOST\_IP   = [HOST\_IP]\\
ROUTER\_IP = [ROUTER\_IP]\\
PUBLIC\_IP = IP pública\\
DEST\_NAME = Google\\
DEST\_IP   = [DEST\_IP]\\
PROTOCOL   = [PROTOCOL]\\
PORT       = [PORT]\\
MODE       = Bridged
\end{terminal}
\end{exampleblock}
\column{.56\textwidth}
\begin{block}{Prompt final}
\scriptsize
\texttt{Crea un diagrama de secuencia del tráfico cuando abro google.com en Firefox desde una VM Ubuntu. Usa la ficha proporcionada. Muestra la resolución DNS, la salida desde VM\_IP hacia ROUTER\_IP, la traducción a PUBLIC\_IP, la conexión PROTOCOL a DEST\_IP:PORT y la respuesta. Indica junto a cada dato el comando usado para obtenerlo.}
\end{block}
\begin{alertblock}{Precisión del modo puente}
\scriptsize Incluye \texttt{HOST\_IP} como contexto de la Mac, pero no como salto: en modo \texttt{Bridged}, la VM participa directamente en la red local.
\end{alertblock}
\end{columns}
\end{frame}
