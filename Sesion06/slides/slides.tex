\begin{frame}{Los permisos responden quién puede hacer qué}
\begin{block}{Idea central}
\small Ubuntu protege cada archivo indicando qué acciones puede realizar el \textbf{propietario}, el \textbf{grupo} y el \textbf{resto de usuarios}.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{exampleblock}{¿Quién?}
\small
\texttt{u}: propietario.\par\medskip
\texttt{g}: grupo.\par\medskip
\texttt{o}: otros usuarios.
\end{exampleblock}
\column{.49\textwidth}
\begin{exampleblock}{¿Qué puede hacer?}
\small
\texttt{r}: leer.\par\medskip
\texttt{w}: modificar.\par\medskip
\texttt{x}: ejecutar o atravesar un directorio.
\end{exampleblock}
\end{columns}
\begin{alertblock}{Principio práctico}
\small Concede solo el permiso necesario para la tarea. Dar más permisos no corrige la causa de un problema.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{ls -l} permite leer los permisos de izquierda a derecha}
\begin{exampleblock}{Archivo antes de hacerlo ejecutable}
\begin{terminal}\footnotesize
rgap@ubuntu:\textasciitilde/Desktop\$ ls -l saludo.py\\
-rw-rw-r-- 1 rgap rgap 0 Aug 27 15:34 saludo.py
\end{terminal}
\end{exampleblock}
\begin{columns}[T,onlytextwidth]
\column{.22\textwidth}
\begin{block}{\texttt{-}}
\small Es un archivo regular. Una \texttt{d} indicaría directorio.
\end{block}
\column{.24\textwidth}
\begin{block}{\texttt{rw-}}
\small El propietario puede leer y modificar; todavía no puede ejecutar.
\end{block}
\column{.24\textwidth}
\begin{block}{\texttt{rw-}}
\small El grupo \texttt{rgap} puede leer y modificar el archivo.
\end{block}
\column{.24\textwidth}
\begin{block}{\texttt{r--}}
\small Los demás usuarios solo pueden leerlo.
\end{block}
\end{columns}
\end{frame}

\begin{frame}{El script necesita una línea de intérprete y permiso \texttt{x}}
\begin{columns}[T,onlytextwidth]
\column{.55\textwidth}
\begin{exampleblock}{Contenido de \texttt{saludo.py}}
\begin{terminal}\footnotesize
\#!/usr/bin/env python3\\
usuario = input('¿Cómo te llamas? ')\\
print(f'Hola, \{usuario\}')
\end{terminal}
\end{exampleblock}
\column{.41\textwidth}
\begin{block}{La primera línea decide quién lo interpreta}
\small \texttt{\#!/usr/bin/env python3} indica que Ubuntu debe buscar Python 3 y usarlo para ejecutar el archivo.
\end{block}
\end{columns}
\begin{alertblock}{Primer intento}
\begin{terminal}
\$ ./saludo.py\\
bash: ./saludo.py: Permiso denegado
\end{terminal}
\small El archivo existe, pero el propietario aún no posee permiso de ejecución.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{chmod u+x} concede ejecución solo al propietario}
\begin{columns}[T,onlytextwidth]
\column{.57\textwidth}
\begin{exampleblock}{Cambio y comprobación}
\begin{terminal}\footnotesize
\$ chmod u+x saludo.py\\
\$ ls -l saludo.py\\
-rwxrw-r-- 1 rgap rgap 157 Aug 27 15:41 saludo.py\\
\$ ./saludo.py\\
¿Cómo te llamas? Ana\\
Hola, Ana. El script ya tiene permiso de ejecución.
\end{terminal}
\end{exampleblock}
\column{.39\textwidth}
\begin{block}{Cómo se lee}
\small
\texttt{u}: propietario.\par\medskip
\texttt{+}: agrega un permiso.\par\medskip
\texttt{x}: permite ejecución directa.
\end{block}
\end{columns}
\begin{alertblock}{Diferencia importante}
\small \texttt{python3 saludo.py} puede funcionar sin \texttt{x} porque ejecutas Python y este lee el archivo. \texttt{./saludo.py} sí demuestra el permiso de ejecución.
\end{alertblock}
\end{frame}

\begin{frame}{La forma simbólica explica mejor la intención}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{exampleblock}{Cambios concretos}
\begin{terminal}
\$ chmod u+x saludo.py\\
\$ chmod g-w datos.txt\\
\$ chmod o-r secreto.txt
\end{terminal}
\small Cada comando expresa quién cambia, si agrega o quita, y qué permiso afecta.
\end{exampleblock}
\column{.49\textwidth}
\begin{exampleblock}{Forma numérica}
\small \texttt{r=4}, \texttt{w=2}, \texttt{x=1}. La suma se calcula para propietario, grupo y otros.
\begin{terminal}
\$ chmod 750 tarea.sh\\
rwx r-x ---
\end{terminal}
\end{exampleblock}
\end{columns}
\begin{alertblock}{Evita \texttt{chmod 777}}
\small Permitir lectura, escritura y ejecución a todos suele ocultar el problema y ampliar innecesariamente el acceso.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{systemd} mantiene disponibles los servicios}
\begin{block}{Tres piezas relacionadas}
\small Un \textbf{servicio} realiza una tarea en segundo plano. Un archivo \texttt{.service} describe cómo ejecutarlo. \texttt{systemd} aplica esa descripción y \texttt{systemctl} permite consultarla o controlarla.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{exampleblock}{Consultar no modifica}
\begin{terminal}
\$ systemctl status mysql\\
\$ systemctl is-active mysql
\end{terminal}
\end{exampleblock}
\column{.49\textwidth}
\begin{exampleblock}{Controlar requiere privilegios}
\begin{terminal}
\$ sudo systemctl start mysql\\
\$ sudo systemctl enable mysql
\end{terminal}
\end{exampleblock}
\end{columns}
\begin{alertblock}{Dos preguntas diferentes}
\small \texttt{active} responde “¿está funcionando ahora?”. \texttt{enabled} responde “¿está configurado para iniciar con Ubuntu?”.
\end{alertblock}
\end{frame}

\begin{frame}{Lista los archivos de servicio conocidos por Ubuntu}
\begin{exampleblock}{Comando solicitado para el laboratorio}
\begin{terminal}\footnotesize
\$ sudo systemctl list-unit-files --type service --all\\
UNIT FILE                    STATE     PRESET\\
mysql.service                enabled   enabled\\
ssh.service                  disabled  enabled\\
systemd-journald.service     static    -
\end{terminal}
\end{exampleblock}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Qué muestra}
\small Los archivos de unidad instalados y su estado de habilitación. No indica por sí solo cuáles están ejecutándose.
\end{block}
\column{.48\textwidth}
\begin{block}{Sobre \texttt{sudo} y \texttt{--all}}
\small La consulta normalmente funciona sin \texttt{sudo}; \texttt{--all} hace explícito que no se filtrará por estado.
\end{block}
\end{columns}
\end{frame}

\begin{frame}{El estado del archivo explica cómo puede iniciarse}
\begin{description}[\texttt{disabled}]\small
  \item[\texttt{enabled}] Tiene enlaces de arranque y normalmente se iniciará al alcanzar su objetivo.
  \item[\texttt{disabled}] Existe, pero no quedó configurado para iniciar automáticamente.
  \item[\texttt{static}] No se habilita directamente; otra unidad puede activarlo como dependencia.
  \item[\texttt{masked}] Está bloqueado y no puede iniciarse hasta retirar la máscara.
  \item[\texttt{PRESET}] Es otra columna: muestra la política recomendada por el paquete o la distribución.
\end{description}
\begin{alertblock}{No confundas estado de archivo con estado actual}
\small Un servicio puede estar \texttt{active} y \texttt{disabled}, o estar \texttt{inactive} y \texttt{enabled}.
\end{alertblock}
\end{frame}

\begin{frame}{Node atiende solicitudes y MySQL mantiene los datos}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{block}{Aplicación Node}
\small Publica una API en el puerto \texttt{8000}. Node.js no crea un servicio genérico \texttt{node.service}; nosotros definiremos \texttt{mi-api.service}.
\end{block}
\column{.49\textwidth}
\begin{block}{Servidor MySQL}
\small Mantiene disponible el motor de base de datos. En Ubuntu, el paquete \texttt{mysql-server} suele administrarse como \texttt{mysql.service}.
\end{block}
\end{columns}
\begin{exampleblock}{Preparación del laboratorio}
\begin{terminal}\footnotesize
\$ sudo apt update\\
\$ sudo apt install nodejs mysql-server -y\\
\$ node --version\\
\$ systemctl status mysql --no-pager
\end{terminal}
\end{exampleblock}
\begin{alertblock}{Comprueba el nombre real}
\small Si usas MariaDB u otro paquete, el nombre puede cambiar. Búscalo antes con \texttt{systemctl list-unit-files '*mysql*' '*mariadb*'}.
\end{alertblock}
\end{frame}

\begin{frame}{La API comprueba si MySQL responde en el puerto \texttt{3306}}
\begin{columns}[T,onlytextwidth]
\column{.61\textwidth}
\begin{exampleblock}{Idea principal de \texttt{app.js}}
\begin{terminal}\footnotesize
const http = require('http');\\
const net = require('net');\\
net.createConnection(\\
\quad \{ host: '127.0.0.1', port: 3306 \}\\
);\\
server.listen(8000, '127.0.0.1');
\end{terminal}
\end{exampleblock}
\column{.35\textwidth}
\begin{block}{Respuesta de salud}
\small Si MySQL acepta la conexión, la API devuelve estado \texttt{200}. Si no responde, devuelve \texttt{503}.
\end{block}
\begin{alertblock}{Alcance}
\small Aquí verificamos disponibilidad del motor; las consultas SQL y credenciales se trabajarán por separado.
\end{alertblock}
\end{columns}

\begin{frame}{La unidad \texttt{mi-api.service} relaciona Node con MySQL}
\begin{columns}[T,onlytextwidth]
\column{.64\textwidth}
\begin{exampleblock}{Archivo de unidad}
\begin{terminal}\scriptsize
\lbrack Unit\rbrack\\
Description=API Node de la Sesion 06\\
Requires=mysql.service\\
After=network.target mysql.service\\
\lbrack Service\rbrack\\
Type=simple\\
User=rgap\\
WorkingDirectory=/home/rgap/Escritorio/curso\_cloud/sesion06/demo\\
ExecStart=/usr/bin/node app.js\\
Restart=on-failure\\
\lbrack Install\rbrack\\
WantedBy=multi-user.target
\end{terminal}
\end{exampleblock}
\column{.32\textwidth}
\begin{block}{Cómo se interpreta}
\small
\texttt{Requires}: declara la dependencia.\par\medskip
\texttt{After}: ordena el inicio.\par\medskip
\texttt{ExecStart}: ejecuta Node.\par\medskip
\texttt{WantedBy}: permite habilitar el arranque.
\end{block}
\end{columns}
\begin{alertblock}{Importante}
\scriptsize Ajusta \texttt{User} y \texttt{WorkingDirectory} a tu cuenta. \texttt{After=mysql.service} ordena el inicio, pero no garantiza que MySQL ya acepte conexiones; por eso comprobamos el puerto.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{enable --now} inicia y habilita los servicios}
\begin{columns}[T,onlytextwidth]
\column{.56\textwidth}
\begin{exampleblock}{Instala la unidad e inicia ambos servicios}
\begin{terminal}\footnotesize
\$ sudo install -m 644 mi-api.service \\
\quad /etc/systemd/system/mi-api.service\\
\$ sudo systemctl daemon-reload\\
\$ sudo systemctl enable --now \\
\quad mysql.service mi-api.service
\end{terminal}
\end{exampleblock}
\column{.40\textwidth}
\begin{exampleblock}{Comprueba}
\begin{terminal}\footnotesize
\$ systemctl is-active \\
\quad mysql mi-api\\
active\\
active\\
\$ curl 127.0.0.1:8000\\
\{"node":"activo",\\
 "mysql":"alcanzable"\}
\end{terminal}
\end{exampleblock}
\end{columns}
\small\textbf{Resultado:} MySQL y la API quedan activos ahora y habilitados para el próximo inicio de Ubuntu.
\end{frame}

\begin{frame}{SSH crea una terminal remota cifrada}
\begin{block}{Qué es}
\small SSH significa \emph{Secure Shell}. Permite autenticarse en otro equipo y ejecutar comandos a través de una conexión protegida.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\begin{exampleblock}{Dónde se usa}
\small Máquinas virtuales, servidores Linux, instancias de Compute Engine y equipos de laboratorio.
\end{exampleblock}
\column{.31\textwidth}
\begin{exampleblock}{Para qué se usa}
\small Administrar servicios, revisar registros, copiar archivos y desplegar aplicaciones.
\end{exampleblock}
\column{.31\textwidth}
\begin{exampleblock}{Qué necesita}
\small Un cliente SSH, un servidor SSH accesible, usuario, dirección y un método de autenticación.
\end{exampleblock}
\end{columns}
\begin{alertblock}{Modelo mental}
\small Tu terminal es el cliente. La VM escucha normalmente en el puerto \texttt{22} y ejecuta allí los comandos autorizados.
\end{alertblock}
\end{frame}

\begin{frame}{GitHub permite trabajar por SSH y esto explica por qué}
\begin{center}
\begin{tikzpicture}[node distance=1cm]
  \node[caja] (cli) {Tu equipo\\ \footnotesize guarda la clave privada};
  \node[cajaverde, right=4.4cm of cli] (srv) {GitHub\\ \footnotesize guarda tu clave pública};
  \draw[flechavuelta] (cli) -- node[above,font=\scriptsize] {SSH, puerto 22} node[below,font=\scriptsize] {\texttt{git@github.com}} (srv);
\end{tikzpicture}
\end{center}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\begin{exampleblock}{Te identifica}
\small Registras tu clave pública en la cuenta. Tu clave privada demuestra quién eres sin enviar contraseña.
\end{exampleblock}
\column{.31\textwidth}
\begin{exampleblock}{Protege el envío}
\small El código y las credenciales viajan por un canal cifrado, aunque uses una red compartida.
\end{exampleblock}
\column{.31\textwidth}
\begin{exampleblock}{Se nota en la URL}
\small \texttt{https://github.com/...} usa la web; \texttt{git@github.com:...} usa SSH.
\end{exampleblock}
\end{columns}
\begin{alertblock}{La misma idea que en una VM}
\small GitHub, tu máquina virtual y un servidor cloud aceptan SSH por lo mismo: identifica al usuario y cifra la conexión. Si pierdes el equipo, se revoca esa clave desde la cuenta sin cambiar nada más.
\end{alertblock}
\end{frame}

\begin{frame}{Servicios y puertos clásicos}
\begin{block}{Por qué importan}
\small Una misma dirección IP atiende varios servicios a la vez. El puerto indica a cuál de ellos llega cada conexión.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Acceso y web}
\small
\texttt{22} SSH: administración remota.\par\smallskip
\texttt{80} HTTP: web sin cifrar.\par\smallskip
\texttt{443} HTTPS: web cifrada.\par\smallskip
\texttt{53} DNS: resolución de nombres.
\end{exampleblock}
\column{.48\textwidth}
\begin{exampleblock}{Datos y aplicaciones}
\small
\texttt{3306} MySQL.\par\smallskip
\texttt{5432} PostgreSQL.\par\smallskip
\texttt{8000} y \texttt{3000}: aplicaciones Node en desarrollo.\par\smallskip
\texttt{25} y \texttt{587} SMTP: correo saliente.
\end{exampleblock}
\end{columns}
\begin{alertblock}{Ya los usaste en esta sesión}
\small \texttt{443} al abrir Google en Firefox, \texttt{3306} cuando \texttt{mi-api} comprueba MySQL y \texttt{8000} al consultar la API. \texttt{ss -ltn} muestra qué puertos escucha tu equipo.
\end{alertblock}
\end{frame}
