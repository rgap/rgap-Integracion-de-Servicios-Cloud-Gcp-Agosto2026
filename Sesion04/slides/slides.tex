\begin{frame}{La máquina virtual ejecuta su propio sistema}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{block}{Tres conceptos de la sesión anterior}
\small
\textbf{Anfitrión:} equipo físico que aporta CPU, memoria, disco y red.\par\medskip
\textbf{Hipervisor:} programa que crea y administra máquinas virtuales.\par\medskip
\textbf{Máquina virtual (MV):} computadora definida mediante software que recibe recursos del anfitrión.
\end{block}
\column{.49\textwidth}
\begin{exampleblock}{Ejemplo completo}
\small
\textbf{Anfitrión:} laptop del estudiante.\par
$\downarrow$\par
\textbf{Hipervisor:} VirtualBox, VMware u otro disponible.\par
$\downarrow$\par
\textbf{Sistema invitado:} sistema operativo que inicia dentro de la MV.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Linux administra recursos; Ubuntu lo entrega listo}
\begin{description}[Distribución]\small
  \item[Sistema operativo] Conjunto de programas que permite utilizar el hardware y ejecutar aplicaciones.
  \item[Kernel] Núcleo del sistema operativo: coordina CPU, memoria, disco, red y dispositivos.
  \item[Unix] Familia de sistemas operativos iniciada en 1969; popularizó usuarios, reglas de acceso (permisos), programas en ejecución (procesos) y un árbol único de archivos.
  \item[Linux] Kernel libre creado en 1991 que sigue muchas ideas de Unix. Osea Unix no era libre.
  \item[Distribución] Sistema listo para instalar: reúne Linux, aplicaciones, instalador, actualizaciones y configuración.
\end{description}
\begin{exampleblock}{Ejemplo}
\small Ubuntu es una \textbf{distribución}; utiliza el \textbf{kernel Linux} para administrar los recursos del equipo.
\end{exampleblock}
\end{frame}

\begin{frame}{Unix y Linux comparten ideas, no el mismo origen}
\begin{center}
\small
\begin{tabular}{>{\bfseries}p{3.1cm}p{3.9cm}p{4.0cm}}
\toprule
Aspecto & Unix & Linux \\
\midrule
Origen & Bell Labs, 1969 & Linus Torvalds, 1991 \\
Qué era al inicio & Sistema operativo completo & Kernel \\
Uso actual & El Unix original ya no se usa; existen sistemas derivados y sistemas que cumplen el estándar UNIX & Kernel de muchas distribuciones GNU/Linux \\
Ejemplos & AIX, HP-UX y macOS & Ubuntu, Debian, Mint y Kali usan Linux \\
\bottomrule
\end{tabular}
\end{center}
\begin{alertblock}{Resultado de la comparación}
\small Linux no es una versión de Unix: es otro kernel que adoptó muchas de sus ideas y formas de trabajo.
\end{alertblock}
\end{frame}

\begin{frame}{Paquete y gestor de paquetes}
\begin{block}{Dos definiciones antes del ejemplo}
\small Un \textbf{paquete} es un archivo que contiene un programa y la información necesaria para instalarlo. Un \textbf{gestor de paquetes} instala, actualiza o elimina esos paquetes.
\end{block}
\begin{center}
\small
\begin{tabular}{>{\bfseries}p{2.6cm}p{4.3cm}p{3.8cm}}
\toprule
Familia & Distribuciones & Paquete y gestor \\
\midrule
Debian & Debian, Ubuntu, Mint, Kali & \texttt{.deb}; \texttt{apt} y \texttt{dpkg} \\
Red Hat & RHEL, Fedora, Rocky, AlmaLinux & \texttt{.rpm}; \texttt{dnf} y \texttt{rpm} \\
Arch / SUSE & Arch, Manjaro, openSUSE & \texttt{pacman} / \texttt{zypper} \\
\bottomrule
\end{tabular}
\end{center}
\small\textbf{Resultado para el laboratorio:} Ubuntu pertenece a la familia Debian; por eso trabajaremos con paquetes \texttt{.deb} y el gestor \texttt{apt}.
\end{frame}

\begin{frame}{Ubuntu recibe programas en archivos \texttt{.deb}}
\begin{block}{Definiciones necesarias}
\small Un \textbf{archivo \texttt{.deb}} es el formato de paquete usado por Debian y Ubuntu. Los \textbf{metadatos} describen el paquete. Una \textbf{dependencia} es otro paquete necesario. La \textbf{arquitectura} indica el tipo de equipo compatible.
\end{block}
\begin{columns}[c,onlytextwidth]
\column{.44\textwidth}
\includegraphics[width=\linewidth]{paquete-deb.png}
\column{.50\textwidth}
\begin{exampleblock}{Así se ve el nombre del archivo}
\centering\large\texttt{nano\_7.2\_amd64.deb}
\end{exampleblock}
\small
\begin{tabular}{>{\bfseries}p{1.7cm}p{3.9cm}}
\texttt{nano} & nombre del programa \\
\texttt{7.2} & versión del paquete \\
\texttt{amd64} & arquitectura de 64 bits \\
\texttt{.deb} & formato de Debian y Ubuntu \\
\end{tabular}
\begin{alertblock}{Qué contiene}
\scriptsize Metadatos —nombre, versión y dependencias—, archivos del programa y documentación.
\end{alertblock}
\end{columns}
\small\textbf{Resultado:} Ubuntu reconoce el archivo \texttt{.deb} porque pertenece a la familia Debian.
\end{frame}

\begin{frame}{Primero inicia la máquina virtual}
\begin{block}{Punto de partida}
\large La MV puede contener una instalación reciente de Ubuntu o un sistema que ya fue utilizado.
\end{block}
\begin{alertblock}{Índice local}
\small Catálogo que Ubuntu guarda en la MV con los paquetes y versiones disponibles en los repositorios configurados.
\end{alertblock}
\begin{enumerate}\setlength\itemsep{9pt}\large
  \item Inicia la MV.
  \item Espera hasta que aparezca el escritorio de Ubuntu.
  \item Como primera tarea, actualiza el índice local.
\end{enumerate}
\begin{exampleblock}{Siguiente acción}
\small La próxima diapositiva muestra la instrucción y el resultado esperado de la actualización.
\end{exampleblock}
\end{frame}

\begin{frame}{Primer inicio: actualiza el índice con \texttt{apt}}
\begin{block}{Definiciones necesarias}
\small Un \textbf{repositorio} es un servidor que almacena paquetes. El \textbf{índice local} es el catálogo de paquetes que Ubuntu guarda en la MV. \texttt{sudo} ejecuta una instrucción con permisos de administración.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.38\textwidth}
\begin{block}{Qué ocurre}
\begin{enumerate}\small
  \item Ejecutas \texttt{sudo apt update}.
  \item \texttt{apt} consulta los repositorios configurados.
  \item Ubuntu renueva su índice local de paquetes.
\end{enumerate}
\end{block}
\column{.58\textwidth}
\begin{exampleblock}{Ejemplo y resultado esperado}
\begin{terminal}
\small sudo apt update\\
Obj:1 archive.ubuntu.com ubuntu InRelease\\
Leyendo lista de paquetes... Hecho\\
Creando árbol de dependencias... Hecho\\
Todos los paquetes están actualizados.
\end{terminal}
\end{exampleblock}
\end{columns}
\small\textbf{Comprobación:} las tareas de lectura y dependencias muestran \texttt{Hecho}. Las líneas pueden variar según Ubuntu y los repositorios configurados.
\end{frame}

\begin{frame}{¿Por qué actualizar el índice antes de instalar?}
\begin{block}{Tres definiciones para el ejemplo}
\small \textbf{Node.js} ejecuta JavaScript fuera del navegador. \textbf{npm} gestiona paquetes para proyectos Node.js. Una \textbf{API} es un servicio que recibe solicitudes y devuelve datos o resultados.
\end{block}
\begin{alertblock}{La razón}
\small \texttt{apt} consulta el índice local para localizar versiones disponibles y sus dependencias. Al actualizarlo primero, evita buscar información antigua antes de una instalación.
\end{alertblock}
\begin{columns}[T,onlytextwidth]
\column{.39\textwidth}
\begin{block}{Para una API}
\begin{enumerate}\small
  \item Actualiza el índice local.
  \item Instala \texttt{nodejs} y \texttt{npm}.
  \item Comprueba ambas versiones.
\end{enumerate}
\end{block}
\footnotesize\textbf{Resultado:} la MV dispone del entorno y del gestor necesarios para preparar una API con Node.js.
\column{.57\textwidth}
\begin{exampleblock}{Instalación y resultado esperado}
\begin{terminal}
\scriptsize sudo apt install nodejs npm\\
¿Desea continuar? [S/n] s\\
Configurando nodejs (...) ...\\
Configurando npm (...) ...\\
node --version\\
v18.x.x\\
npm --version\\
9.x.x
\end{terminal}
\end{exampleblock}
\end{columns}
\end{frame}

\begin{block}{Dos definiciones antes del ejemplo}
\small La \textbf{Terminal} es la aplicación donde escribes instrucciones y lees resultados. La \textbf{shell} es el programa que interpreta cada instrucción y solicita su ejecución.
\end{block}
\begin{columns}[c,onlytextwidth]

\column{.44\textwidth}
\begin{exampleblock}{Resultado completo}

\begin{frame}{Una línea distingue cuatro elementos}
\small El \textbf{prompt} aparece antes de escribir. La \textbf{instrucción} comienza con el comando y puede incluir una opción y un argumento.
\begin{center}
\begin{tikzpicture}[node distance=1.6mm]
  \node[caja,fill=utpGrisClaro,minimum width=4.0cm] (p) {\texttt{rgap@ubuntu:\textasciitilde\$}};
  \node[caja,fill=utpRojo!12,minimum width=2.0cm,right=of p] (c) {\texttt{ls}};
  \node[caja,fill=utpGrisClaro,minimum width=2.0cm,right=of c] (o) {\texttt{-a}};
  \node[caja,fill=utpRojo!12,minimum width=2.4cm,right=of o] (a) {\texttt{/etc}};

  \node[below=4mm of p,text width=3.6cm,align=center,font=\small] {\textbf{Prompt}\\Texto previo que identifica usuario, equipo y ubicación.};
  \node[below=4mm of c,text width=2.4cm,align=center,font=\small] {\textbf{Comando}\\Acción que se solicita: listar.};
  \node[below=4mm of o,text width=2.4cm,align=center,font=\small] {\textbf{Opción}\\Cambia cómo actúa el comando.};
  \node[below=4mm of a,text width=2.7cm,align=center,font=\small] {\textbf{Argumento}\\Dato sobre el que actúa: \texttt{/etc}.};
\end{tikzpicture}
\end{center}
\begin{exampleblock}{Lectura completa}
\small El prompt indica dónde estás. La instrucción \texttt{ls -a /etc} solicita el contenido de \texttt{/etc}, incluidos los elementos ocultos.
\end{exampleblock}
\end{frame}



\begin{frame}{El sistema de archivos organiza todo desde una raíz}
\begin{description}[Ruta relativa]\small
  \item[Sistema de archivos] Forma en que Linux organiza directorios y archivos.
  \item[Directorio] Contenedor que puede guardar archivos y otros directorios.
  \item[Raíz \texttt{/}] Punto inicial del árbol completo.
  \item[Ruta absoluta] Ubicación que comienza en \texttt{/}; no depende del directorio actual.
  \item[Ruta relativa] Ubicación calculada desde el directorio actual.
  \item[Directorio personal] Espacio del usuario, representado también por \texttt{\textasciitilde}.
\end{description}
\begin{exampleblock}{Ejemplo}
\small Absoluta: \texttt{/home/rgap/Documentos} · relativa desde \texttt{/home/rgap}: \texttt{Documentos}
\end{exampleblock}
\end{frame}

\begin{frame}{Cada rama del árbol cumple una función}
\begin{columns}[c,onlytextwidth]
\column{.61\textwidth}
\includegraphics[width=\linewidth]{sistema-archivos.png}
\column{.35\textwidth}
\begin{exampleblock}{Resultado completo}
\small
\texttt{/home}: archivos personales.\par
\texttt{/etc}: configuración.\par
\texttt{/var}: datos variables y registros de eventos (logs).\par
\texttt{/tmp}: archivos temporales.\par
\texttt{/usr/bin}: programas y comandos.\par\medskip
Todos descienden de la raíz \texttt{/}.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{\texttt{pwd}: muestra la ubicación actual}
\begin{block}{Definición}
\large \texttt{pwd} muestra la ruta completa del directorio en el que estás trabajando.
\end{block}
\begin{exampleblock}{Ejemplo y resultado completo}
\begin{terminal}
rgap@ubuntu:\textasciitilde\$ pwd\\
/home/rgap
\end{terminal}
\end{exampleblock}
\begin{alertblock}{Cómo comprobarlo}
\small La salida comienza con \texttt{/}; por eso es una ruta absoluta. En este ejemplo, corresponde al directorio personal de \texttt{rgap}.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{ls -a}: muestra también los elementos ocultos}
\begin{block}{Dos definiciones}
\small \texttt{ls} lista el contenido del directorio actual. La opción \texttt{-a} incluye los nombres ocultos. Un \textbf{repositorio Git} es un directorio cuyo historial y configuración se guardan en el directorio oculto \texttt{.git}.
\end{block}
\begin{exampleblock}{Ejemplo en el directorio de una API}
\begin{terminal}
rgap@ubuntu:\textasciitilde/api-repo\$ ls -a\\
.\quad ..\quad .git\quad .gitignore\quad README.md\\
package.json\quad src
\end{terminal}
\end{exampleblock}
\begin{alertblock}{Resultado}
\small \texttt{.git} confirma que el directorio es un repositorio Git. \texttt{.} representa la ubicación actual y \texttt{..}, la ubicación superior.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{mkdir}: crea un directorio}
\begin{block}{Definición}
\large \texttt{mkdir} crea un directorio nuevo dentro de la ubicación actual.
\end{block}
\begin{exampleblock}{Ejemplo y comprobación}
\begin{terminal}
rgap@ubuntu:\textasciitilde\$ mkdir api\\
rgap@ubuntu:\textasciitilde\$ ls -a\\
.\quad ..\quad api
\end{terminal}
\end{exampleblock}
\begin{alertblock}{Resultado}
\small \texttt{mkdir api} no imprime una confirmación cuando funciona. El resultado se comprueba porque \texttt{api} aparece en el listado.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{cd}: cambia la ubicación actual}
\begin{block}{Definición}
\large \texttt{cd} cambia la ubicación actual al directorio indicado como argumento.
\end{block}
\begin{exampleblock}{Ejemplo y resultado completo}
\begin{terminal}
rgap@ubuntu:\textasciitilde\$ cd api\\
rgap@ubuntu:\textasciitilde/api\$ pwd\\
/home/rgap/api
\end{terminal}
\end{exampleblock}
\begin{alertblock}{Cómo comprobarlo}
\small El prompt cambia a \texttt{\textasciitilde/api} y \texttt{pwd} confirma la nueva ruta completa.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{cat}: muestra el contenido de un archivo}
\begin{block}{Definición}
\large \texttt{cat} lee un archivo y muestra su contenido como salida.
\end{block}
\begin{exampleblock}{Ejemplo en un repositorio Git}
\begin{terminal}
rgap@ubuntu:\textasciitilde/api-repo\$ cat README.md\\
\# API del curso\\
Servicio de ejemplo con Node.js
\end{terminal}
\end{exampleblock}
\begin{alertblock}{Resultado}
\small La salida coincide con el texto guardado en \texttt{README.md}; el archivo no se modifica.
\end{alertblock}
\end{frame}

\begin{frame}{Otros comandos identifican usuario, equipo y versión}
\begin{columns}[T,onlytextwidth]
\column{.46\textwidth}
\begin{description}[\texttt{cat /etc/os-release}]\small
  \item[\texttt{whoami}] Usuario actual.
  \item[\texttt{hostname}] Nombre del equipo.
  \item[\texttt{cat /etc/os-release}] Distribución y versión.
  \item[\texttt{uname -a}] Kernel, nombre del equipo y arquitectura.
\end{description}
\column{.50\textwidth}
\begin{terminal}
rgap@ubuntu:\textasciitilde\$ whoami\\
rgap\\
rgap@ubuntu:\textasciitilde\$ hostname\\
ubuntu\\
rgap@ubuntu:\textasciitilde\$ uname -a\\
Linux ubuntu 6.8.0 x86\_64 GNU/Linux
\end{terminal}
\end{columns}
\vspace{3mm}
\small\textbf{Resultado:} ya puedes demostrar quién usa la MV, cómo se llama y qué kernel ejecuta.
\end{frame}

\begin{frame}{Práctica 1A: identifica el sistema \textnormal{\small(5 min)}}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{enumerate}\setlength\itemsep{4pt}\small
  \item Ejecuta \texttt{whoami}, \texttt{hostname} y \texttt{cat /etc/os-release}.
  \item Crea primero \texttt{\textasciitilde/curso\_cloud} y después \texttt{sesion04} con \texttt{mkdir}.
\end{enumerate}
\column{.48\textwidth}
\begin{exampleblock}{Resultado completo esperado}
\begin{terminal}
\$ whoami\\
rgap\\
\$ hostname\\
ubuntu\\
\$ cat /etc/os-release\\
NAME="Ubuntu"\\
VERSION="24.04 LTS"\\
\$ mkdir \textasciitilde/curso\_cloud\\
\$ mkdir \textasciitilde/curso\_cloud/sesion04
\end{terminal}
\end{exampleblock}
\end{columns}
\begin{alertblock}{Seguridad}
\small Comprende cada comando y confirma la ruta. No uses \texttt{sudo} —ejecución con privilegios administrativos— ni comandos de borrado sin saber qué cambiarán.
\end{alertblock}
\end{frame}

\begin{frame}{Práctica 1B: guarda la evidencia \textnormal{\small(5 min)}}
\begin{columns}[T,onlytextwidth]
\column{.43\textwidth}
\begin{enumerate}\setlength\itemsep{6pt}\small
  \item Ingresa con \texttt{cd}.
  \item Confirma la ruta con \texttt{pwd}.
  \item Guarda \texttt{uname -a > evidencia.txt}.
  \item Revisa el archivo con \texttt{cat}.
\end{enumerate}
\column{.53\textwidth}
\begin{exampleblock}{Resultado completo esperado}
\begin{terminal}
\$ cd \textasciitilde/curso\_cloud/sesion04\\
\$ pwd\\
/home/rgap/curso\_cloud/sesion04\\
\$ uname -a > evidencia.txt\\
\$ cat evidencia.txt\\
Linux ubuntu 6.8.0 x86\_64 GNU/Linux
\end{terminal}
\end{exampleblock}
\end{columns}
\vspace{3mm}
\small\textbf{Comprobación:} el archivo \texttt{evidencia.txt} existe en la ruta mostrada y contiene la salida de \texttt{uname -a}.
\end{frame}
