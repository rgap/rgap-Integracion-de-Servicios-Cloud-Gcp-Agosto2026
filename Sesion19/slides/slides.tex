\begin{frame}{Regiones y zonas: dónde vive cada recurso}
\begin{block}{Dos preguntas distintas}
\small La \textbf{jerarquía} dice \textbf{de quién} es un recurso (su proyecto). La \textbf{región} y la \textbf{zona} dicen \textbf{dónde está} físicamente.
\end{block}
\renewcommand{\arraystretch}{1.3}
\centering\small
\begin{tabular}{@{}p{2.2cm}>{\raggedright\arraybackslash}p{4.8cm}p{2.7cm}>{\raggedright\arraybackslash}p{2.8cm}@{}}
\toprule
\textbf{Alcance} & \textbf{Qué significa} & \textbf{Ejemplo de hoy} & \textbf{Como\ldots}\\
\midrule
\textbf{Global} & Existe en todo el mundo a la vez & La red VPC & la empresa\\
\textbf{Regional} & Vive en una región: una zona geográfica & La subred & su sede en una ciudad\\
\textbf{Zonal} & Vive en una zona: un centro de datos de esa región & La máquina virtual & un edificio de esa sede\\
\bottomrule
\end{tabular}
\par\vspace{2mm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Cómo leer un nombre}
\small \texttt{us-central1} es una \textbf{región}. \texttt{us-central1-a} es una \textbf{zona} de esa región: la letra final la delata.
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{Por eso el lab te las da}
\small Cada lab te indica \textbf{su} región y \textbf{su} zona. Si usas otras, la tarea no se pone en verde.
\end{alertblock}

\begin{frame}{De la oficina a la nube: Google Cloud VPC}
\begin{block}{Definición}
\small \textbf{VPC} (\emph{Virtual Private Cloud}, nube privada virtual): la \textbf{red privada} de tu proyecto dentro de Google Cloud. Es la misma idea de la red local, pero \textbf{sin cables}: se arma con clics o comandos.
\end{block}
\renewcommand{\arraystretch}{1.3}
\centering\small
\begin{tabular}{@{}p{3.0cm}p{4.5cm}p{5.4cm}@{}}
\toprule
 & \textbf{En la oficina} & \textbf{En Google Cloud}\\
\midrule
\textbf{La red} & Cables, switch y router & Una \textbf{red VPC}: se crea en segundos\\
\textbf{La subred} & Un piso o un área & Un pedazo de la VPC en \textbf{una región}\\
\textbf{El gateway} & El router que compras e instalas & Google lo pone \textbf{solo} en cada subred\\
\textbf{Los equipos} & PCs e impresoras & Máquinas virtuales\\
\bottomrule
\end{tabular}
\par\vspace{2mm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{El gateway, sin trabajo}
\small Dos subredes de la \textbf{misma VPC} ya se comunican: Google hace de gateway entre ellas. Tú no configuras nada.
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{Lo mismo, fuera de Google}
\small \textbf{VPC} en AWS y \textbf{Virtual Network} en Azure. La idea es la misma en todas las nubes.
\end{alertblock}
\end{columns}
\end{frame}

\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{Un solo sentido}
\small De automático a personalizado \textbf{sí} se puede cambiar. De regreso, \textbf{no}.

\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{A y B: misma red, países distintos}
\small Se hablan \textbf{directo}, por dentro, como si estuvieran en el mismo cuarto.
\end{block}
\column{.48\textwidth}
\begin{block}{C y D: redes distintas, misma ciudad}
\small \textbf{No} se hablan por dentro. Solo si cada una sale con su dirección pública.
\end{block}
\end{columns}
\par\vspace{1mm}
\begin{exampleblock}{Una comparación}
\small Dos anexos de la \textbf{misma empresa} se llaman gratis aunque estén en ciudades distintas. Dos \textbf{empresas distintas}, aunque compartan edificio, tienen que llamarse por el número público.
\end{exampleblock}
\end{frame}

\begin{frame}{Dirección IP interna y externa}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{IP interna}
\small La tiene \textbf{toda} máquina virtual. Sirve para hablar \textbf{dentro} de su red. Google la asigna sola.
\end{block}
\begin{block}{IP externa}
\small Es \textbf{opcional}. Sirve para que te vean desde \textbf{Internet}. La \emph{URL del sitio} de la sesión anterior era esto.
\end{block}
\column{.48\textwidth}
\begin{exampleblock}{Una comparación}
\small La interna es tu \textbf{anexo}: solo funciona dentro de la oficina. La externa es el \textbf{número público}: cualquiera puede marcarlo.
\end{exampleblock}
\begin{alertblock}{La externa puede cambiar}
\small \textbf{Efímera:} si apagas y prendes la VM, te dan otra.\par
\textbf{Estática:} la reservas y es siempre la misma.
\end{alertblock}
\end{columns}
\par\vspace{1mm}
\begin{block}{Lo que muestra la demostración del curso}
\small Apagan una VM y la vuelven a prender: la IP \textbf{interna quedó igual}; la \textbf{externa cambió}.
\end{block}
\end{frame}

\begin{frame}{Firewall: quién puede pasar}
\begin{center}
\begin{tikzpicture}[font=\small,
  caja/.style={draw=utpRojo,rounded corners=4pt,minimum width=2.6cm,minimum height=1.0cm,align=center}]
  \node[caja,fill=white] (i) at (0,0) {\textbf{Alguien}\\\footnotesize quiere entrar};
  \node[caja,fill=utpRojo!30] (f) at (5.0,0) {\textbf{Firewall}\\\footnotesize revisa su lista};
  \node[caja,fill=utpRojo!10] (v) at (10.0,0) {\textbf{Tu VM}};
  \draw[-latex,thick] (i) -- (f);
  \draw[-latex,thick] (f) -- node[above,font=\footnotesize] {si está permitido} (v);
\end{tikzpicture}
\end{center}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Las dos reglas que siempre están}
\small Lo que \textbf{entra}: todo \textbf{bloqueado}.\par
Lo que \textbf{sale}: todo \textbf{permitido}.\par\smallskip
Tú agregas reglas para \textbf{dejar entrar} lo que necesitas.
\end{block}
\column{.48\textwidth}
\begin{exampleblock}{Una regla dice tres cosas}
\small
\textbf{Qué:} protocolo y puerto (SSH, puerto 22).\par
\textbf{Desde dónde:} qué direcciones.\par
\textbf{A quién:} todas las VMs o solo algunas.
\end{exampleblock}
\end{columns}
\par\vspace{1mm}
\begin{alertblock}{Una comparación}
\small El \textbf{portero} del condominio con su lista de invitados: si no estás en la lista, no entras. Tú, que vives ahí, sales cuando quieres.
\end{alertblock}
\end{frame}

\begin{frame}{Lo más importante del lab: ¿quién se puede hablar?}
\begin{block}{El experimento}
\small Entras a \texttt{mynet-us-vm} y le haces \texttt{ping} a las otras tres máquinas: primero a su IP \textbf{externa}, luego a su IP \textbf{interna}.
\end{block}
\renewcommand{\arraystretch}{1.3}
\centering\small
\begin{tabular}{@{}p{4.3cm}p{3.6cm}p{2.3cm}p{2.3cm}@{}}
\toprule
\textbf{Destino} & \textbf{Dónde está} & \textbf{IP externa} & \textbf{IP interna}\\
\midrule
\texttt{mynet-notus-vm} & misma red, otra región & Responde & \textbf{Responde}\\
\texttt{managementnet-us-vm} & otra red, misma zona & Responde & \textbf{No llega}\\
\texttt{privatenet-us-vm} & otra red, misma zona & Responde & \textbf{No llega}\\
\bottomrule
\end{tabular}
\par\vspace{2mm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Qué significa}
\small Por dentro, lo que importa es estar en la \textbf{misma red}. La distancia no importa.
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{La conclusión del lab}
\small Cada red VPC está \textbf{aislada} de las demás. Para unirlas hay que configurarlo aparte.
\end{alertblock}
\end{columns}
\par\vspace{1mm}
{\raggedright\small \textbf{¿Por qué la IP externa sí responde?} Porque va por Internet, y tus reglas de firewall dejan entrar el \texttt{ping}.\par}
\end{frame}

\begin{frame}{El lab: tres ideas para no perderte}
\renewcommand{\arraystretch}{1.45}
\centering\small
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{3.6cm}>{\raggedright\arraybackslash}p{5.3cm}>{\raggedright\arraybackslash}p{4.0cm}@{}}
\toprule
\textbf{La idea} & \textbf{Dónde la ves en el lab} & \textbf{Se parece a}\\
\midrule
\textbf{Sin red no hay VM} & Borras la red y ya no te deja crear máquinas & Pedir un anexo donde no hay central telefónica\\
\textbf{Para entrar hace falta permiso} & Una regla de firewall deja abrir la terminal solo a las VMs con la etiqueta \texttt{iap-gce} & Un sticker de ``visita autorizada''\\
\textbf{El \texttt{ping} se lee fácil} & \texttt{0\% packet loss}: respondió. \texttt{100\% packet loss}: no llegó & Gritar ``¿me escuchas?''\\
\bottomrule
\end{tabular}
\par\vspace{3mm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Lo haces de dos maneras}
\small Una red con \textbf{clics} y otra con \textbf{comandos}. Hacen lo mismo: son las dos puertas de la Sesión 17.
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{Si algo falla}
\small Copia los nombres, la región y la zona \textbf{tal cual} los da el lab. Y sigue la guía del lab paso a paso: ahí está cada clic.
\end{alertblock}
\end{columns}
\end{frame}
