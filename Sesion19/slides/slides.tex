\begin{frame}{Regiones y zonas: dónde vive cada recurso}
\begin{block}{Dos preguntas distintas}
\small La \textbf{jerarquía} dice \textbf{de quién} es un recurso (su proyecto). La \textbf{región} y la \textbf{zona} dicen \textbf{dónde está} físicamente.
\end{block}
\renewcommand{\arraystretch}{1.3}
\centering\small

\textbf{Alcance} & \textbf{Qué significa} & \textbf{Ejemplo de hoy} & \textbf{Como\ldots}\\

\textbf{Regional} & Vive en una región: una zona geográfica & La subred & su sede en una ciudad\\

\begin{frame}{De la oficina a la nube: Google Cloud VPC}
\begin{block}{Definición}
\small \textbf{VPC} (\emph{Virtual Private Cloud}, nube privada virtual): la \textbf{red privada} de tu proyecto dentro de Google Cloud. Es la misma idea de la red local, pero \textbf{sin cables}: se arma con clics o comandos.
\end{block}
\renewcommand{\arraystretch}{1.3}
\centering\small
\begin{tabular}{@{}p{3.0cm}p{4.5cm}p{5.4cm}@{}}
\toprule
 & \textbf{En la oficina} & \textbf{En Google Cloud}\\
\midrule
\textbf{La red} & Cables, switch y router & Una \textbf{red VPC}: se crea en segundos\\
\textbf{La subred} & Un piso o un área & Un pedazo de la VPC en \textbf{una región}\\
\textbf{El gateway} & El router que compras e instalas & Google lo pone \textbf{solo} en cada subred\\
\textbf{Los equipos} & PCs e impresoras & Máquinas virtuales\\
\bottomrule
\end{tabular}
\par\vspace{2mm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{El gateway, sin trabajo}
\small Dos subredes de la \textbf{misma VPC} ya se comunican: Google hace de gateway entre ellas. Tú no configuras nada.
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{Lo mismo, fuera de Google}
\small \textbf{VPC} en AWS y \textbf{Virtual Network} en Azure. La idea es la misma en todas las nubes.
\end{alertblock}
\end{columns}
\end{frame}

\separador{Parte 2: Acceso privado a Google y Cloud NAT}{Máquinas sin dirección pública que igual pueden trabajar}

\begin{frame}{El punto de partida: una VM sin IP externa}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Por qué se hace}
\small Una máquina \textbf{sin IP externa} no se ve desde Internet: nadie de afuera puede tocarla. Es lo más \textbf{seguro} para una base de datos o un servidor interno.
\end{block}
\begin{exampleblock}{Una comparación}
\small Un empleado con \textbf{anexo pero sin número público}: nadie de la calle puede llamarlo directo.
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{El problema}
\small La puerta queda cerrada en \textbf{los dos sentidos}. Esa máquina tampoco puede:\par\smallskip
\textbf{1.} Usar los \textbf{servicios de Google}, como Cloud Storage.\par
\textbf{2.} \textbf{Salir a Internet} para descargar actualizaciones.
\end{alertblock}
\begin{block}{Lo que sí puede}
\small Hablar con las máquinas de \textbf{su misma red}, por su IP interna.
\end{block}
\end{columns}
\par\vspace{1mm}
\begin{block}{Las dos soluciones de hoy}
\small Para el problema 1: \textbf{Acceso privado a Google}. Para el problema 2: \textbf{Cloud NAT}.
\end{block}
\end{frame}

\begin{frame}{Acceso privado a Google}
\begin{center}
\begin{tikzpicture}[font=\small,
  caja/.style={draw=utpRojo,rounded corners=4pt,minimum width=2.9cm,minimum height=1.0cm,align=center}]
  \node[caja,fill=white] (v) at (0,0) {\textbf{VM}\\\footnotesize sin IP externa};
  \node[caja,fill=utpRojo!30] (g) at (8.4,0) {\textbf{Servicios de Google}\\\footnotesize Cloud Storage y otros};
  \draw[-latex,thick] (v) -- node[above,font=\footnotesize] {por dentro de la red de Google} (g);
\end{tikzpicture}
\end{center}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Qué es}
\small Un \textbf{interruptor} de la subred. Al encenderlo, sus máquinas sin IP externa pueden usar los \textbf{servicios de Google}.
\end{block}
\begin{block}{Dónde se enciende}
\small En la \textbf{subred}, no en la máquina. Vale para todas las VMs de esa subred.
\end{block}
\column{.48\textwidth}
\begin{exampleblock}{Una comparación}
\small Un \textbf{pasillo interno} entre tu oficina y el almacén de la misma empresa: llegas sin salir a la calle.
\end{exampleblock}
\begin{alertblock}{Lo que no hace}
\small No te lleva al \textbf{resto de Internet}. Solo a Google.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Cloud NAT}
\begin{center}
\begin{tikzpicture}[font=\small,
  caja/.style={draw=utpRojo,rounded corners=4pt,minimum width=2.7cm,minimum height=1.0cm,align=center}]
  \node[caja,fill=white] (v) at (0,0) {\textbf{VM}\\\footnotesize sin IP externa};
  \node[caja,fill=utpRojo!30] (n) at (4.3,0) {\textbf{Cloud NAT}\\\footnotesize presta una IP pública};

  \draw[-latex,thick] ([yshift=2.2mm]v.east) -- node[above,font=\footnotesize] {sale} ([yshift=2.2mm]n.west);
  \draw[-latex,thick] ([yshift=2.2mm]n.east) -- node[above,font=\footnotesize] {sale con la IP prestada} ([yshift=2.2mm]i.west);
  \draw[-latex,thick,dashed] ([yshift=-2.2mm]i.west) -- node[below,font=\footnotesize] {nadie entra por su cuenta} ([yshift=-2.2mm]n.east);
\end{tikzpicture}
\end{center}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Qué es}
\small Un servicio que deja a las máquinas sin IP externa \textbf{salir a Internet}. La respuesta vuelve, pero nadie de afuera puede \textbf{empezar} una conexión hacia ellas.
\end{block}
\begin{block}{Una palabra nueva}
\small \textbf{NAT:} \emph{traducción de direcciones}. Cambia la IP interna por una pública al salir.
\end{block}
\column{.48\textwidth}
\begin{exampleblock}{Una comparación}
\small La \textbf{central telefónica} de la empresa: llamas afuera y se ve el número de la empresa. Desde afuera nadie marca directo a tu anexo.
\end{exampleblock}
\begin{alertblock}{Ya lo usas en casa}
\small Tu \textbf{router wifi} hace NAT: por eso tu celular navega y nadie de Internet entra a él.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{¿Cuál necesito?}
\renewcommand{\arraystretch}{1.4}
\centering\small
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{5.0cm}>{\raggedright\arraybackslash}p{4.3cm}p{3.5cm}@{}}

Descargar actualizaciones de Internet & \textbf{Cloud NAT} & la red y una región\\
Que yo entre a su terminal & \textbf{IAP}, como en el lab 1 & una regla de firewall\\
Que el público la visite & Una \textbf{IP externa} & la propia VM\\
\bottomrule
\end{tabular}
\par\vspace{3mm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Se pueden usar juntos}
\small En el lab 2 enciendes \textbf{los dos}: primero el Acceso privado y después Cloud NAT.
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{¿Y cómo entro sin IP externa?}
\small Con \textbf{IAP}: Google te abre un túnel seguro hasta la VM. Es la opción \texttt{--tunnel-through-iap} que ya usaste.
\end{alertblock}
\end{columns}
\end{frame}

\begin{block}{Los comandos}
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize]
export MY_BUCKET=nombre-de-tu-bucket
gcloud storage cp gs://cloud-training/gcpnet/private/access.svg gs://$MY_BUCKET
gcloud storage cp gs://$MY_BUCKET/*.svg .
\end{lstlisting}
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.52\textwidth}
\begin{block}{Paso a paso}
\small
\textbf{1.} Crea un bucket \textbf{Multirregión} con un nombre único.\par
\textbf{2.} En Cloud Shell, ejecuta las líneas 1 y 2: la imagen queda en tu bucket.\par
\textbf{3.} Entra a \texttt{vm-internal} y ejecuta la línea 3: \textbf{se cuelga}. Corta con \texttt{Ctrl + C}.\par
\textbf{4.} Edita la subred \texttt{privatenet-us}: Acceso privado a Google \textbf{Activo}.\par
\textbf{5.} Repite la línea 3 en la VM: \textbf{ahora funciona}.
\end{block}
\column{.44\textwidth}
\begin{exampleblock}{Cómo leerlo}
\small \texttt{export}: guarda el nombre. \texttt{cp}: copia. \texttt{gs://}: es un bucket.
\end{exampleblock}
\begin{alertblock}{Error típico}
\small La variable \texttt{MY\_BUCKET} vive en \textbf{Cloud Shell}. Dentro de la VM no existe: hay que repetir el \texttt{export} ahí.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Lab 2, tareas 3 y 4: configura Cloud NAT}
\begin{columns}[T,onlytextwidth]
\column{.52\textwidth}
\begin{block}{Paso a paso}
\small
\textbf{1.} Dentro de \texttt{vm-internal}, ejecuta \texttt{sudo apt-get update}. Solo avanza con lo que es de Google; el resto \textbf{se cuelga}. Corta con \texttt{Ctrl + C} y sal con \texttt{exit}.\par\smallskip
\textbf{2.} Busca \textbf{Servicios de red} $\rightarrow$ \textbf{Cloud NAT} $\rightarrow$ \textbf{Comenzar}.\par\smallskip
\textbf{3.} Nombre: \texttt{nat-config}. Red: \texttt{privatenet}. Región: la del lab.\par\smallskip
\textbf{4.} Cloud Router: \textbf{Crear router nuevo}, \texttt{nat-router}. Clic en \textbf{Crear}.\par\smallskip
\textbf{5.} Espera el estado \textbf{En ejecución} y repite el comando en la VM: \textbf{ahora funciona}.
\end{block}
\column{.44\textwidth}
\begin{exampleblock}{¿Qué hace \texttt{apt-get update}?}
\small Le pregunta a Internet qué actualizaciones hay para Linux. Es una prueba simple de ``¿puedo salir?''.
\end{exampleblock}
\begin{block}{¿Y el Cloud Router?}
\small La pieza sobre la que trabaja Cloud NAT. Solo le pones nombre.
\end{block}
\begin{alertblock}{Ten paciencia}
\small Cloud NAT puede tardar \textbf{hasta 3 minutos} en hacer efecto. Si falla al primer intento, espera y repite.
\end{alertblock}
\end{columns}
\par\vspace{1mm}
{\raggedright\small \textbf{Tarea 4:} edita \texttt{nat-config} y, en \textbf{Registros}, elige \textbf{Traducción y errores}. Repite el comando y mira las líneas nuevas en el \textbf{Explorador de registros}.\par}
\end{frame}
