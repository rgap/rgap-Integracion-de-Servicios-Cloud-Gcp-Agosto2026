\begin{frame}{Linux administra recursos; Ubuntu lo entrega listo}
\begin{description}[Distribución]\small
  \item[Sistema operativo] Conjunto de programas que permite utilizar el hardware y ejecutar aplicaciones.
  \item[Kernel] Núcleo del sistema operativo: coordina CPU, memoria, disco, red y dispositivos.
  \item[Unix] Familia de sistemas operativos iniciada en 1969; popularizó usuarios, reglas de acceso (permisos), programas en ejecución (procesos) y un árbol único de archivos.
  \item[Linux] Kernel libre creado en 1991 que sigue muchas ideas de Unix. Osea Unix no era libre.
  \item[Distribución] Sistema listo para instalar: reúne Linux, aplicaciones, instalador, actualizaciones y configuración.
\end{description}
\begin{exampleblock}{Ejemplo}
\small Ubuntu es una \textbf{distribución}; utiliza el \textbf{kernel Linux} para administrar los recursos del equipo.
\end{exampleblock}
\end{frame}

\begin{frame}{Unix y Linux comparten ideas, no el mismo origen}
\begin{center}

\begin{tabular}{>{\bfseries}p{3.1cm}p{3.9cm}p{4.0cm}}
\toprule
Aspecto & Unix & Linux \\
\midrule
Origen & Bell Labs, 1969 & Linus Torvalds, 1991 \\
Qué era al inicio & Sistema operativo completo & Kernel \\
Uso actual & El Unix original ya no se usa; existen sistemas derivados y sistemas que cumplen el estándar UNIX & Kernel de muchas distribuciones GNU/Linux \\

\begin{frame}{Paquete y gestor de paquetes}
\begin{block}{Dos definiciones antes del ejemplo}
\small Un \textbf{paquete} es un archivo que contiene un programa y la información necesaria para instalarlo. Un \textbf{gestor de paquetes} instala, actualiza o elimina esos paquetes.
\end{block}
\begin{center}
\small
\begin{tabular}{>{\bfseries}p{2.6cm}p{4.3cm}p{3.8cm}}
\toprule
Familia & Distribuciones & Paquete y gestor \\
\midrule
Debian & Debian, Ubuntu, Mint, Kali & \texttt{.deb}; \texttt{apt} y \texttt{dpkg} \\
Red Hat & RHEL, Fedora, Rocky, AlmaLinux & \texttt{.rpm}; \texttt{dnf} y \texttt{rpm} \\
Arch / SUSE & Arch, Manjaro, openSUSE & \texttt{pacman} / \texttt{zypper} \\
\bottomrule
\end{tabular}
\end{center}
\small\textbf{Resultado para el laboratorio:} Ubuntu pertenece a la familia Debian; por eso trabajaremos con paquetes \texttt{.deb} y el gestor \texttt{apt}.
\end{frame}

\begin{frame}{Primer inicio: actualiza el índice con \texttt{apt}}

\begin{enumerate}\small
  \item Ejecutas \texttt{sudo apt update}.

\end{enumerate}
\end{block}

\begin{frame}{¿Por qué actualizar el índice antes de instalar?}
\begin{block}{Tres definiciones para el ejemplo}
\small \textbf{Node.js} ejecuta JavaScript fuera del navegador. \textbf{npm} gestiona paquetes para proyectos Node.js. Una \textbf{API} es un servicio que recibe solicitudes y devuelve datos o resultados.
\end{block}
\begin{alertblock}{La razón}
\small \texttt{apt} consulta el índice local para localizar versiones disponibles y sus dependencias. Al actualizarlo primero, evita buscar información antigua antes de una instalación.
\end{alertblock}
\begin{columns}[T,onlytextwidth]
\column{.39\textwidth}
\begin{block}{Para una API}
\begin{enumerate}\small
  \item Actualiza el índice local.
  \item Instala \texttt{nodejs} y \texttt{npm}.
  \item Comprueba ambas versiones.
\end{enumerate}
\end{block}
\footnotesize\textbf{Resultado:} la MV dispone del entorno y del gestor necesarios para preparar una API con Node.js.
\column{.57\textwidth}
\begin{exampleblock}{Instalación y resultado esperado}

Configurando npm (...) ...\\

v18.x.x\\

9.x.x

\begin{frame}{El sistema de archivos organiza todo desde una raíz}
\begin{description}[Ruta relativa]\small
  \item[Sistema de archivos] Forma en que Linux organiza directorios y archivos.
  \item[Directorio] Contenedor que puede guardar archivos y otros directorios.
  \item[Raíz \texttt{/}] Punto inicial del árbol completo.
  \item[Ruta absoluta] Ubicación que comienza en \texttt{/}; no depende del directorio actual.
  \item[Ruta relativa] Ubicación calculada desde el directorio actual.
  \item[Directorio personal] Espacio del usuario, representado también por \texttt{\textasciitilde}.
\end{description}
\begin{exampleblock}{Ejemplo}
\small Absoluta: \texttt{/home/rgap/Documentos} · relativa desde \texttt{/home/rgap}: \texttt{Documentos}
\end{exampleblock}
\end{frame}

\begin{frame}{\texttt{pwd}: muestra la ubicación actual}
\begin{block}{Definición}
\large \texttt{pwd} muestra la ruta completa del directorio en el que estás trabajando.
\end{block}
\begin{exampleblock}{Ejemplo y resultado completo}
\begin{terminal}
rgap@ubuntu:\textasciitilde\$ pwd\\
/home/rgap
\end{terminal}
\end{exampleblock}
\begin{alertblock}{Cómo comprobarlo}
\small La salida comienza con \texttt{/}; por eso es una ruta absoluta. En este ejemplo, corresponde al directorio personal de \texttt{rgap}.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{ls -a}: muestra también los elementos ocultos}
\begin{block}{Dos definiciones}
\small \texttt{ls} lista el contenido del directorio actual. La opción \texttt{-a} incluye los nombres ocultos. Un \textbf{repositorio Git} es un directorio cuyo historial y configuración se guardan en el directorio oculto \texttt{.git}.
\end{block}
\begin{exampleblock}{Ejemplo en el directorio de una API}
\begin{terminal}
rgap@ubuntu:\textasciitilde/api-repo\$ ls -a\\
.\quad ..\quad .git\quad .gitignore\quad README.md\\
package.json\quad src
\end{terminal}
\end{exampleblock}
\begin{alertblock}{Resultado}
\small \texttt{.git} confirma que el directorio es un repositorio Git. \texttt{.} representa la ubicación actual y \texttt{..}, la ubicación superior.
\end{alertblock}
\end{frame}

\begin{frame}{El resultado de \texttt{ip a} muestra dos interfaces}
\begin{columns}[T,onlytextwidth]
\column{.54\textwidth}
\begin{exampleblock}{Líneas principales del resultado}
\begin{terminal}
rgap@ubuntu:\textasciitilde\$ ip a\\
1: lo\\
\quad inet 127.0.0.1/8 scope host lo\\
2: ens160\\
\quad inet 192.168.35.151/24\\
\quad scope global dynamic ens160
\end{terminal}
\end{exampleblock}
\column{.42\textwidth}
\begin{description}[\texttt{ens160}]\small
  \item[\texttt{ip a}] Lista interfaces y direcciones.
  \item[\texttt{lo}] Interfaz interna del propio Ubuntu.
  \item[\texttt{ens160}] Conecta la MV con la red.
  \item[\texttt{inet}] Presenta una dirección IPv4: cuatro números separados por puntos.
  \item[\texttt{/24}] Indica la parte que identifica la red local.
\end{description}
\end{columns}
\begin{alertblock}{Qué significa este resultado}
\small \texttt{state UP}: \texttt{ens160} está activa. \texttt{dynamic}: su dirección fue asignada automáticamente. \texttt{UNKNOWN} en \texttt{lo} no es un error: no hay una conexión externa que comprobar porque \texttt{lo} funciona dentro de Ubuntu y no depende de un cable ni del adaptador virtual.
\end{alertblock}
\scriptsize Otros campos: \texttt{link/ether} = dirección del adaptador; \texttt{mtu}, \texttt{qdisc} y \texttt{qlen} = transmisión de datos; \texttt{inet6} = dirección IP más larga; \texttt{valid\_lft} = tiempo de validez.
\end{frame}

\begin{frame}{Tu MV muestra dos interfaces: \texttt{lo} y \texttt{ens160}}
\begin{block}{Una es interna y la otra conecta con la red}
\small El resultado de \texttt{ip a} presenta dos interfaces con funciones diferentes.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{exampleblock}{\texttt{lo}: comunicación interna}
\small
Dirección: \texttt{127.0.0.1/8}.\par
Ubuntu la usa para comunicarse consigo mismo. No conecta la MV con otros equipos.
\end{exampleblock}
\column{.49\textwidth}
\begin{exampleblock}{\texttt{ens160}: comunicación con la red}
\small
Dirección: \texttt{192.168.35.151/24}.\par
Es la interfaz que permite que la MV se comunique con la red externa.
\end{exampleblock}
\end{columns}
\begin{alertblock}{Lectura correcta del resultado}
\small Aunque \texttt{ip a} muestra dos interfaces, solo \texttt{ens160} conecta la MV con la red externa. \texttt{lo} funciona únicamente dentro de Ubuntu.
\end{alertblock}
\end{frame}

\begin{frame}{Ruta, puerta de enlace y DNS describen el camino}
\begin{description}[Puerta de enlace]\small
  \item[Ruta] Regla que indica por dónde debe enviarse el tráfico hacia un destino.
  \item[Puerta de enlace] Router que permite salir de la red local; también se denomina \emph{gateway}.
  \item[DNS] Servicio que traduce nombres, como \texttt{example.com}, a direcciones IP.
\end{description}
\begin{exampleblock}{Ejemplo completo}
\small En este laboratorio, la MV \texttt{192.168.35.151/24} sale mediante \emph{Share with my Mac}. La Mac usa \texttt{192.168.100.140/24} en el Wi-Fi y su puerta de enlace es \texttt{192.168.100.1}. El DNS resuelve el nombre del sitio antes de iniciar la conexión.
\end{exampleblock}
\end{frame}

\begin{frame}{La MV llega a internet a través de la Mac}
\begin{block}{NAT conecta las dos redes}
\small \textbf{NAT} significa traducción de direcciones de red. En \emph{Share with my Mac}, la Mac recibe el tráfico de la red virtual y lo envía por su conexión Wi-Fi usando su propia dirección.
\end{block}

\small El DNS convierte el nombre del sitio en una IP. La solicitud sale por \texttt{ens160}, la Mac aplica NAT, el Wi-Fi la entrega al router y el router la envía a internet. La respuesta regresa por el camino inverso hasta la MV.

\begin{frame}{Alias y función reducen instrucciones repetidas}
\begin{columns}[c,onlytextwidth]
\column{.56\textwidth}
\includegraphics[width=\linewidth]{alias-funcion.png}
\column{.40\textwidth}
\begin{block}{Alias}
\small Nombre corto que representa \textbf{un comando} con sus opciones. Ejemplo: \texttt{o} puede representar \texttt{xdg-open .}.
\end{block}
\begin{exampleblock}{Función}
\small Nombre que agrupa \textbf{uno o varios comandos} ejecutados en orden. Forma: \texttt{nombre() \{ comandos; \}}.
\end{exampleblock}
\begin{alertblock}{Duración en esta práctica}
\small Ambos existen solo en la terminal actual. Al cerrarla, desaparecen.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Bash y Zsh son shells; \texttt{\$SHELL} indica la configurada}
\begin{block}{Variable de entorno}
\small Dato con nombre que la shell conserva para describir la sesión. \texttt{SHELL} guarda la ubicación completa de la shell configurada. El comando \texttt{echo} muestra un texto o el valor de una variable.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{exampleblock}{Bash en Ubuntu}
\begin{terminal}
rgap@ubuntu:\textasciitilde\$ echo \$SHELL\\
/bin/bash
\end{terminal}
\end{exampleblock}
\column{.49\textwidth}
\begin{exampleblock}{Zsh en otra cuenta}
\begin{terminal}
rel@equipo:\textasciitilde\$ echo \$SHELL\\
/bin/zsh
\end{terminal}
\end{exampleblock}
\end{columns}
\small\textbf{Resultado:} ambas son shells; la salida permite reconocer cuál está configurada.
\end{frame}

\begin{frame}{\texttt{xdg-open .} abre la carpeta actual en Ubuntu}
\begin{block}{Dos elementos antes de crear el alias}
\small \texttt{xdg-open} solicita a Ubuntu abrir un elemento con la aplicación gráfica predeterminada. El punto \texttt{.} representa el directorio actual.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.50\textwidth}
\begin{exampleblock}{Comando completo}
\begin{terminal}
rgap@ubuntu:\textasciitilde\$ pwd\\
/home/rgap\\
rgap@ubuntu:\textasciitilde\$ xdg-open .
\end{terminal}
\end{exampleblock}
\column{.46\textwidth}
\begin{alertblock}{Resultado visible}
\small Ubuntu abre la aplicación \textbf{Archivos} y muestra el contenido de \texttt{/home/rgap}, que era el directorio actual.
\end{alertblock}
\end{columns}
\vspace{4mm}
\small\textbf{Idea del alias:} escribir \texttt{o} será más rápido que repetir \texttt{xdg-open .}.
\end{frame}

\begin{frame}{Práctica 2: crea un alias y una función \textnormal{\small(12 min)}}
\begin{columns}[T,onlytextwidth]
\column{.46\textwidth}
\begin{enumerate}\setlength\itemsep{4pt}\small
  \item Confirma la shell con \texttt{echo \$SHELL}.
  \item Crea \texttt{alias o='xdg-open .'}.
  \item Ejecuta \texttt{o} y confirma que se abra \textbf{Archivos}.
  \item Crea \texttt{info() \{ whoami; hostname; pwd; \}}.
  \item Ejecuta \texttt{info}.
\end{enumerate}
\column{.50\textwidth}
\begin{exampleblock}{Resultado completo esperado}
\begin{terminal}\$ echo \$SHELL\\
/bin/bash\\
\$ alias o='xdg-open .'\\
\$ o\\
Se abre la carpeta actual en Archivos\\
\$ info() \{ whoami; hostname; pwd; \}\\
\$ info\\
rgap\\
ubuntu\\
/home/rgap/curso\_cloud/sesion03
\end{terminal}
\end{exampleblock}
\end{columns}
\begin{alertblock}{Comprobación final}
\small \texttt{o} abre la carpeta actual en Archivos; \texttt{info} devuelve usuario, equipo y ruta, en ese orden.
\end{alertblock}
\end{frame}

\begin{frame}{\texttt{.bashrc} conserva alias y funciones}
\begin{block}{Archivo de configuración de Bash}
\small \texttt{.bashrc} es un archivo oculto ubicado en el directorio personal. Bash lee sus instrucciones cada vez que inicia una terminal interactiva.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{exampleblock}{Solo en la terminal actual}
\small Si creas \texttt{o} e \texttt{info} directamente en la terminal, desaparecen cuando la cierras.
\end{exampleblock}
\column{.49\textwidth}
\begin{exampleblock}{Guardados en \texttt{.bashrc}}
\small Si escribes allí el alias y la función, Bash vuelve a crearlos al abrir cada terminal nueva.
\end{exampleblock}
\end{columns}
\begin{alertblock}{Dos comandos que utilizaremos}
\small \texttt{nano \textasciitilde/.bashrc} abre el archivo en un editor de texto. \texttt{source \textasciitilde/.bashrc} vuelve a leerlo sin cerrar la terminal actual.
\end{alertblock}
\end{frame}

\begin{frame}{Práctica 3: guarda los atajos en \texttt{.bashrc} \textnormal{\small(12 min)}}
\begin{columns}[T,onlytextwidth]
\column{.43\textwidth}
\begin{enumerate}\setlength\itemsep{4pt}\small
  \item Abre el archivo con \texttt{nano \textasciitilde/.bashrc}.
  \item Agrega el alias \texttt{o} y la función \texttt{info} al final.
  \item Guarda con \texttt{Ctrl+O}, confirma con \texttt{Enter} y sal con \texttt{Ctrl+X}.
  \item Ejecuta \texttt{source \textasciitilde/.bashrc}.
  \item Prueba \texttt{o} e \texttt{info}.
\end{enumerate}
\column{.53\textwidth}
\begin{exampleblock}{Contenido y resultado esperado}
\begin{terminal}
\$ nano \textasciitilde/.bashrc\\
alias o='xdg-open .'\\
info() \{ whoami; hostname; pwd; \}\\
\$ source \textasciitilde/.bashrc\\
\$ o\\
Se abre la carpeta actual en Archivos\\
\$ info\\
rgap\\
ubuntu\\
/home/rgap/curso\_cloud/sesion03
\end{terminal}
\end{exampleblock}
\end{columns}
\small\textbf{Comprobación:} abre una terminal nueva y ejecuta \texttt{o} e \texttt{info}; ambos deben seguir disponibles.
\end{frame}
