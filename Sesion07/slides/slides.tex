\begin{frame}{Un servicio es un programa que espera peticiones}
\begin{block}{Idea central}
\small El \textbf{cliente} inicia la conversación y pide algo. El \textbf{servidor} no hace nada hasta que alguien le pide: permanece a la espera, atiende y vuelve a esperar.
\end{block}
\begin{center}
\begin{tikzpicture}[node distance=1cm]
  \node[caja] (cli) {Cliente\\ \footnotesize \texttt{curl}, Firefox, tu webapp de frontend};
  \node[cajaverde, right=4.6cm of cli] (srv) {Servidor\\ \footnotesize servidor HTTP, FTP, tu API};
  \draw[flecha] ([yshift=2.5mm]cli.east) -- node[above,font=\scriptsize] {petición} ([yshift=2.5mm]srv.west);
  \draw[flecha] ([yshift=-2.5mm]srv.west) -- node[below,font=\scriptsize] {respuesta} ([yshift=-2.5mm]cli.east);
\end{tikzpicture}
\end{center}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Servicio o \emph{daemon}}
\small Un programa que el sistema mantiene corriendo en segundo plano, sin ventana ni terminal propia, administrado por \texttt{systemd}.
\end{exampleblock}
\column{.48\textwidth}
\begin{exampleblock}{Un mismo equipo hace ambos papeles}
\small Tu VM es servidor cuando publica una API y cliente cuando consulta \texttt{google.com}.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{El socket es el punto donde se atiende la conexión}
\begin{block}{Qué lo define}
\small Un socket queda identificado por tres datos: \textbf{protocolo} (TCP o UDP), \textbf{dirección IP} y \textbf{puerto}. Por eso una misma IP puede atender muchos servicios a la vez.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.50\textwidth}
\begin{exampleblock}{Los cuatro pasos del servidor}
\begin{terminal}\footnotesize
socket()  crea el punto de conexión\\
bind()    lo fija a IP y puerto\\
listen()  abre la cola de espera\\
accept()  atiende a un cliente
\end{terminal}
\end{exampleblock}
\column{.46\textwidth}
\begin{block}{La dirección de escucha importa}
\small \texttt{127.0.0.1} solo acepta clientes de la propia máquina.
\end{block}
\begin{block}{Un puerto, un servicio}
\small Si dos programas piden el mismo puerto, el segundo falla con \texttt{Address already in use}.
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Teoría: HTTP es texto acordado que viaja sobre TCP}
\begin{block}{Por qué se puede escribir a mano}
\small HTTP no tiene nada de mágico: el cliente envía líneas de texto y el servidor devuelve otras líneas de texto. Lo único obligatorio es respetar el formato: una línea en blanco separa las cabeceras del contenido.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Lo que envía el cliente}
\begin{terminal}\footnotesize
GET / HTTP/1.1\\
Host: 127.0.0.1:8080\\
\textit{(línea en blanco)}
\end{terminal}
\small Método, ruta y versión; luego las cabeceras.
\end{exampleblock}
\column{.48\textwidth}
\begin{exampleblock}{Lo que responde el servidor}
\begin{terminal}\footnotesize
HTTP/1.1 200 OK\\
Content-Type: text/plain\\
Content-Length: 41\\
\textit{(línea en blanco)}\\
Hola desde un servidor...
\end{terminal}
\small Código de estado, cabeceras y contenido.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Cliente HTTP con sockets: \texttt{cliente\_http.py}}
\begin{columns}[T,onlytextwidth]
\column{.56\textwidth}
\begin{exampleblock}{Núcleo del script}
\begin{terminal}\scriptsize
import socket\\
c = socket.socket(socket.AF\_INET,\\
\quad socket.SOCK\_STREAM)\\
c.connect(('127.0.0.1', 8080))\\
c.sendall(PETICION.encode())\\
respuesta = c.recv(1024)\\
c.close()\\
print(respuesta.decode())
\end{terminal}
\small Es lo mismo que hace \texttt{curl}, escrito a mano.
\end{exampleblock}
\column{.40\textwidth}
\begin{exampleblock}{El cliente hace menos pasos}
\begin{terminal}\footnotesize
socket()   crea el punto\\
connect()  busca al servidor\\
sendall()  envía la petición\\
recv()     lee la respuesta
\end{terminal}
\small No hay \texttt{bind()} ni \texttt{listen()}: el cliente no espera a nadie.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Servidor de base de datos simulado: \texttt{servidor\_mysql.py}}
\begin{columns}[T,onlytextwidth]
\column{.56\textwidth}
\begin{exampleblock}{El mismo esqueleto, otra respuesta}
\begin{terminal}\scriptsize
TABLA = \lbrack (1, 'Ana', 'ana@utp.edu.pe'), ... \rbrack\\
~\\
s.bind(('127.0.0.1', 3307))\\
s.listen(5)\\
while True:\\
\quad cli, dir = s.accept()\\
\quad consulta = cli.recv(1024).decode()\\
\quad cli.sendall(responder(consulta).encode())\\
\quad cli.close()
\end{terminal}
\end{exampleblock}
\column{.40\textwidth}
\begin{exampleblock}{Qué cambia frente al de HTTP}
\small Solo el puerto y lo que se devuelve: en vez de una página, filas de una tabla.
\end{exampleblock}
\begin{block}{Puerto 3307}
\small Se usa \texttt{3307} para no chocar con un MySQL real, que ocupa el \texttt{3306}.
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Cliente de base de datos: \texttt{cliente\_mysql.py}}
\begin{columns}[T,onlytextwidth]
\column{.52\textwidth}
\begin{exampleblock}{Conecta, consulta y lee}
\begin{terminal}\scriptsize
c.connect(('127.0.0.1', 3307))\\
c.sendall('SELECT * FROM usuarios'.encode())\\
respuesta = c.recv(4096)\\
c.close()\\
print(respuesta.decode())
\end{terminal}
\end{exampleblock}
\column{.44\textwidth}
\begin{exampleblock}{Resultado}
\begin{terminal}\footnotesize
\$ python3 cliente\_mysql.py\\
Consulta enviada:\\
SELECT * FROM usuarios\\
1 | Ana | ana@utp.edu.pe\\
2 | Luis | luis@utp.edu.pe\\
3 | Sara | sara@utp.edu.pe
\end{terminal}
\end{exampleblock}
\end{columns}
\begin{block}{La misma secuencia que el cliente HTTP}
\small \texttt{connect()}, \texttt{sendall()}, \texttt{recv()}. Cambia el puerto y el texto que se envía; el mecanismo es idéntico. Un MySQL real usa un formato binario más elaborado, pero por debajo también es esto.
\end{block}
\end{frame}

\begin{frame}{Cliente, servidor y servicio no son lo mismo}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\begin{exampleblock}{Cliente}
\small Quien inicia la conexión y pide algo. Termina cuando obtiene su respuesta.
\end{exampleblock}
\column{.31\textwidth}
\begin{exampleblock}{Servidor}
\small Quien escucha en un puerto y responde. Necesita estar activo antes que el cliente.
\end{exampleblock}
\column{.31\textwidth}
\begin{exampleblock}{Servicio}
\small Cualquier programa que el sistema mantiene en segundo plano para cumplir una función.
\end{exampleblock}
\end{columns}
\begin{block}{No todo servicio es un servidor}
\small Un servidor siempre escucha peticiones; un servicio no tiene por qué hacerlo. \texttt{cron} ejecuta tareas a cierta hora, un servicio de respaldo copia archivos de madrugada y \texttt{systemd-timesyncd} ajusta el reloj: ninguno abre un puerto ni espera clientes, y aun así los tres son servicios.
\end{block}
\begin{exampleblock}{El firewall es un servicio, no un servidor}
\small Nadie se conecta al firewall. Se ejecuta en segundo plano revisando el tráfico que entra y decidiendo si dejarlo pasar. Por eso no tiene cliente ni responde peticiones: solo filtra.
\end{exampleblock}
\end{frame}

\begin{frame}{Teoría: un firewall decide qué conexiones pasan}
\begin{block}{Cómo razona}
\small Recorre una lista de \textbf{reglas} en orden. Si alguna coincide con el origen y el puerto, aplica su acción. Si ninguna coincide, aplica la \textbf{política por defecto}.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\begin{exampleblock}{Política por defecto: denegar}
\small Se cierra todo y se abre solo lo necesario. Es el criterio recomendado en servidores.
\end{exampleblock}
\column{.31\textwidth}
\begin{exampleblock}{Qué mira cada regla}
\small Dirección de origen, puerto de destino, protocolo y la acción: permitir o denegar.
\end{exampleblock}
\column{.31\textwidth}
\begin{exampleblock}{Escuchar no es ser accesible}
\small El programa puede estar activo y aun así el firewall impedir que lleguen las conexiones.
\end{exampleblock}
\end{columns}
\end{frame}

\quad \{'origen': '0.0.0.0/0',\\

\end{exampleblock}

\begin{frame}{UFW aplica en el sistema las reglas que simulamos}
\begin{columns}[T,onlytextwidth]

\begin{frame}{Casos usuales de uso de un firewall}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Cerrar la base de datos al exterior}
\small La app se conecta desde la propia máquina o desde la red interna; el \texttt{3306} nunca debe estar abierto a Internet.
\end{exampleblock}

\small En un servidor web se dejan abiertos \texttt{80} y \texttt{443}, y se cierra todo lo demás.
\end{exampleblock}

\small Permitir el \texttt{22} solo desde la IP de la oficina o de la VPN, no desde cualquier origen.
\end{exampleblock}

\begin{exampleblock}{Bloquear un origen abusivo}
\small Banear la IP que está intentando adivinar contraseñas o saturando el servidor.
\end{exampleblock}
\begin{exampleblock}{Cerrar puertos de desarrollo}
\small \texttt{3000}, \texttt{8000} y paneles internos sirven mientras desarrollas; en producción no deben quedar expuestos.
\end{exampleblock}
\begin{exampleblock}{Separar ambientes}
\small Que el servidor de pruebas solo acepte conexiones del equipo de desarrollo.
\end{exampleblock}
\end{columns}

\begin{frame}{Una misma IP atiende varios servicios a la vez}
\begin{block}{Por qué importan}
\small El puerto indica a cuál de esos servicios llega cada conexión. Los números que siguen son \textbf{estándares}: la convención que todo el mundo espera. Pero, como ya vieron, nada impide levantar un servidor en el puerto que quieran —lo hicimos en el \texttt{8080} y en el \texttt{3307}—; solo que entonces el cliente debe indicarlo.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Acceso y web}
\small
\texttt{22} SSH: administración remota.\par\smallskip
\texttt{80} HTTP: web sin cifrar.\par\smallskip
\texttt{443} HTTPS: web cifrada.\par\smallskip
\texttt{53} DNS: resolución de nombres.
\end{exampleblock}
\column{.48\textwidth}
\begin{exampleblock}{Datos y aplicaciones}
\small
\texttt{3306} MySQL.\par\smallskip
\texttt{5432} PostgreSQL.\par\smallskip
\texttt{8000} y \texttt{3000}: aplicaciones Node o Python en desarrollo.\par\smallskip
\texttt{25} y \texttt{587} SMTP: correo saliente.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Qué hace un analista de seguridad ante un puerto abierto}
\begin{block}{Cada puerto abierto es superficie de ataque}
\small El trabajo no es cerrarlo todo, sino justificar uno por uno: qué servicio escucha ahí, quién necesita alcanzarlo y desde dónde.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{1. Inventariar}
\begin{terminal}\footnotesize
\$ sudo ss -ltnp
\end{terminal}
\small Qué escucha, en qué dirección y qué programa lo abrió.
\end{exampleblock}
\begin{exampleblock}{2. Mirar desde fuera}
\begin{terminal}\footnotesize
\$ nmap -sV VM\_IP
\end{terminal}
\small Ve lo mismo que vería un atacante, incluida la versión del servicio.
\end{exampleblock}
\column{.48\textwidth}
\begin{exampleblock}{3. Preguntar por cada puerto}
\small ¿Debe ser accesible desde Internet, solo desde la red interna o solo desde la propia máquina?
\end{exampleblock}
\begin{exampleblock}{4. Reducir y vigilar}
\small Cerrar lo que no se justifica, restringir por origen, mantener el servicio actualizado y revisar los registros de acceso.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Un servidor DNS traduce nombres en direcciones IP}
\begin{block}{Qué problema resuelve}
\small Las personas recordamos \texttt{google.com}; las máquinas necesitan una IP. El DNS es el servicio que convierte lo primero en lo segundo, y escucha en el puerto \texttt{53}. Si falla, el navegador no encuentra el sitio aunque el servidor esté funcionando.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.46\textwidth}
\begin{exampleblock}{Registros frecuentes}
\small
\texttt{A}: apunta un nombre a una IPv4.\par\smallskip
\texttt{AAAA}: lo mismo, con IPv6.\par\smallskip
\texttt{CNAME}: alias de otro nombre.\par\smallskip
\texttt{MX}: a qué servidor va el correo.\par\smallskip
\texttt{TXT}: notas y verificaciones de dominio.
\end{exampleblock}
\column{.50\textwidth}
\begin{block}{Lo que harás en la práctica}
\small Al contratar un hosting o crear una VM te entregan una \textbf{IP pública}. Para que tu dominio apunte ahí, entras al panel del proveedor del dominio y editas el \textbf{registro A}, poniendo esa IP.
\end{block}
\begin{exampleblock}{Y el \texttt{www}}
\small Suele resolverse con un \texttt{CNAME} de \texttt{www.tudominio.com} hacia \texttt{tudominio.com}, para no repetir la IP en dos sitios.
\end{exampleblock}
\end{columns}
\end{frame}

{\color{utpRojo}\rule{8.5cm}{1.2pt}}
\end{center}

\begin{frame}{Los permisos responden quién puede hacer qué}
\begin{block}{Idea central}
\small Ubuntu protege cada archivo indicando qué acciones puede realizar el \textbf{propietario}, el \textbf{grupo} y el \textbf{resto de usuarios}. Un servidor lo usan varias personas y varios servicios: los permisos evitan que un programa cualquiera lea las credenciales de otro.
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{exampleblock}{¿Quién?}
\small
\texttt{u}: propietario.\par\medskip
\texttt{g}: grupo.\par\medskip
\texttt{o}: otros usuarios.
\end{exampleblock}
\column{.49\textwidth}
\begin{exampleblock}{¿Qué puede hacer?}
\small
\texttt{r}: leer.\par\medskip
\texttt{w}: modificar.\par\medskip
\texttt{x}: ejecutar o atravesar un directorio.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{\texttt{ls -l} se lee de izquierda a derecha}
\begin{exampleblock}{Los archivos del proyecto}
\begin{terminal}\footnotesize
rgap@ubuntu:\textasciitilde/proyecto\$ ls -l\\
-rw-rw-r-- 1 rgap rgap  223 Sep  1 13:51 .env\\
-rw-rw-r-- 1 rgap rgap 1339 Sep  1 13:50 servidor\_http.py
\end{terminal}
\end{exampleblock}
\begin{columns}[T,onlytextwidth]
\column{.22\textwidth}
\begin{block}{\texttt{-}}
\small Archivo regular. Una \texttt{d} indicaría directorio.
\end{block}
\column{.24\textwidth}
\begin{block}{\texttt{rw-}}
\small El propietario lee y modifica; todavía no ejecuta.
\end{block}
\column{.24\textwidth}
\begin{block}{\texttt{rw-}}
\small El grupo \texttt{rgap} lee y modifica.
\end{block}
\column{.24\textwidth}
\begin{block}{\texttt{r--}}
\small Los demás solo leen. Aquí está el problema del \texttt{.env}.
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Caso 1: un \texttt{.py} necesita intérprete y permiso \texttt{x}}
\begin{columns}[T,onlytextwidth]
\column{.52\textwidth}
\begin{exampleblock}{Primera línea del script}
\begin{terminal}\footnotesize
\#!/usr/bin/env python3\\
import socket\\
...
\end{terminal}
\small Le indica a Ubuntu qué programa debe interpretarlo.
\end{exampleblock}
\begin{block}{Por qué \texttt{python3 archivo.py} sí funciona}
\small Ahí ejecutas Python, y Python solo \emph{lee} el archivo. \texttt{./archivo.py} exige que el archivo mismo sea ejecutable.
\end{block}
\column{.44\textwidth}
\begin{exampleblock}{Antes y después}
\begin{terminal}\footnotesize
\$ ./servidor\_http.py\\
bash: Permiso denegado\\
\$ chmod u+x servidor\_http.py\\
\$ ./servidor\_http.py\\
Escuchando en \\
http://127.0.0.1:8080
\end{terminal}
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Caso 2: el \texttt{.env} guarda secretos y debe cerrarse}
\begin{columns}[T,onlytextwidth]
\column{.50\textwidth}
\begin{exampleblock}{Qué contiene}
\begin{terminal}\footnotesize
DB\_USER=app\_web\\
DB\_PASSWORD=cambia-esta-clave\\
API\_KEY=clave-de-ejemplo-no-real
\end{terminal}
\small Credenciales que dan acceso real a la base de datos y a servicios de pago.
\end{exampleblock}
\column{.46\textwidth}
\begin{exampleblock}{Ciérralo al propietario}
\begin{terminal}\footnotesize
\$ chmod 600 .env\\
\$ ls -l .env\\
-rw------- 1 rgap rgap 223 .env
\end{terminal}
\small \texttt{600}: el propietario lee y escribe; nadie más entra.
\end{exampleblock}
\end{columns}
\begin{block}{El permiso no reemplaza al \texttt{.gitignore}}
\small \texttt{chmod} protege el archivo en esta máquina. Para que no llegue al repositorio, además debe estar listado en \texttt{.gitignore}.
\end{block}
\end{frame}

\begin{frame}{Forma simbólica y forma numérica}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{exampleblock}{Simbólica: dices qué cambiar}
\begin{terminal}
\$ chmod u+x app.py\\
\$ chmod o-r .env
\end{terminal}
\small La primera agrega ejecución al propietario. La segunda quita la lectura a los demás.
\end{exampleblock}
\column{.49\textwidth}
\begin{exampleblock}{Numérica: \texttt{r=4}, \texttt{w=2}, \texttt{x=1}}
\begin{terminal}
4+2+1 = 7  rwx\\
4+2   = 6  rw-\\
4     = 4  r--
\end{terminal}
\small Un dígito por cada grupo: propietario, grupo y otros.
\end{exampleblock}
\end{columns}
\begin{exampleblock}{Los tres que verás casi siempre}
\begin{terminal}\footnotesize
chmod 600 .env          rw- --- ---   solo el propietario\\
chmod 644 index.html    rw- r-- r--   todos leen, solo tú editas\\
chmod 777 archivo       rwx rwx rwx   todos hacen todo: evítalo
\end{terminal}
\small \texttt{777} no arregla la causa de un error de permisos; solo amplía quién puede provocarlo.
\end{exampleblock}
\end{frame}

\begin{frame}{Práctica: sitio en React generado con IA y sus permisos \textnormal{\small(20 min)}}
\begin{block}{1. Pide el proyecto a \texttt{agy} o a Warp}
\scriptsize\texttt{Crea un sitio estático muy simple en React con Vite que muestre una lista de usuarios traída de https://jsonplaceholder.typicode.com/users. Una sola pantalla, sin router ni librerías extra. Agrega un .env con VITE\_API\_URL, un script iniciar.sh que ejecute npm install y npm run dev, y un .gitignore que excluya .env y node\_modules. Antes de escribir nada, explica el plan.}
\end{block}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{2. Revisa lo que quedó}
\begin{terminal}\footnotesize
\$ ls -la\\
\$ cat .gitignore\\
\$ ./iniciar.sh\\
bash: Permiso denegado
\end{terminal}
\end{exampleblock}
\column{.48\textwidth}
\begin{exampleblock}{3. Corrige los permisos}
\begin{terminal}\footnotesize
\$ chmod u+x iniciar.sh\\
\$ chmod 600 .env\\
\$ ls -l iniciar.sh .env
\end{terminal}
\end{exampleblock}
\end{columns}
\begin{block}{Comprobación y entrega}
\small \texttt{iniciar.sh} debe empezar con \texttt{-rwx} y \texttt{.env} con \texttt{-rw-------}. El sitio debe abrir y mostrar los usuarios. Explica en una línea por qué cada archivo lleva ese permiso y no otro.
\end{block}
\end{frame}
