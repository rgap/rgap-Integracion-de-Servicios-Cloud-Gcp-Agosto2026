% 5 ---------------------------------------------------------------------
\begin{frame}{«Servidor» significa tres cosas distintas}
\begin{columns}[T,onlytextwidth]
\column{.315\textwidth}
\begin{block}{1. Un equipo}
\small Una computadora encendida las 24 horas, casi siempre sin pantalla ni teclado, guardada en un centro de datos.
\end{block}
\column{.315\textwidth}
\begin{block}{2. Un programa}
\small Software que espera solicitudes y responde. Ejemplos reales: \marca{Apache} \marca{Nginx} \marca{Node.js + Express}
\end{block}
\column{.315\textwidth}
\begin{block}{3. Un rol}
\small El que responde en una conversación. El mismo equipo puede ser cliente en otra conversación.
\end{block}
\end{columns}
\vspace{0.2cm}
\analogia{En un restaurante: el local es el \emph{equipo}, el cocinero es el \emph{programa} y «atender pedidos» es la función propia del \emph{rol} de «servidor». Cuando decimos «servidor» casi siempre nos referimos al rol.}
\vspace{0.1cm}
\small Por eso tu laptop puede comportarse como servidor: basta que ejecute un programa que atienda solicitudes.
\end{frame}

\begin{block}{Para encontrarse hace falta una dirección}
\small Cada equipo de la red tiene una \textbf{dirección IP} (por ejemplo \texttt{192.168.1.20}), igual que una dirección postal: sin ella el mensaje no sabría a dónde llegar.

\end{exampleblock}
\end{columns}

% 9 ---------------------------------------------------------------------
\begin{frame}{Usuarios, grupos y permisos responden tres preguntas}
\begin{description}[Recurso compartido]\small
  \item[Usuario] Persona o proceso autorizado para utilizar el sistema operativo. Responde: \emph{¿quién?}
  \item[Grupo] Conjunto de usuarios que comparte permisos. Responde: \emph{¿quiénes reciben la misma regla?}
  \item[Permiso] Autorización para leer, modificar o ejecutar un recurso. Responde: \emph{¿qué puede hacer?}
  \item[Recurso compartido] Archivo, carpeta o dispositivo disponible mediante la red. Responde: \emph{¿sobre qué?}
\end{description}
\vspace{0.15cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Regla de acceso}
\small Una regla relaciona \textbf{usuario o grupo + acción permitida + recurso}.
\end{block}
\column{.48\textwidth}
\begin{exampleblock}{Una regla concreta}
\small El grupo \textbf{Vendedores} puede \textbf{modificar} el recurso \texttt{/stock}.
\end{exampleblock}
\end{columns}
\vspace{0.1cm}
\analogia{Es la tarjeta de un hotel: abre tu habitación y el gimnasio, pero no la oficina de administración.}
\end{frame}

% 10 ---------------------------------------------------------------------
\begin{frame}{Un rol agrupa permisos para una responsabilidad}
\concepto{Rol}{Conjunto de permisos asociado con una función dentro del sistema.}
\begin{center}
\small
\begin{tabular}{>{\bfseries}p{2.7cm}p{7.4cm}}
\toprule
Rol & Acciones permitidas \\
\midrule
Cliente & Registrar pedidos \\
Vendedor & Modificar el stock \\
Administrador & Registrar pedidos, modificar el stock y gestionar usuarios \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.15cm}
\begin{block}{¿Por qué roles y no permisos uno por uno?}
\small Si mañana entra un nuevo vendedor, solo se le asigna el rol: no hay que volver a escribir sus permisos. Y si el rol cambia, cambian todos a la vez.
\end{block}
\vspace{0.05cm}
\small La \textbf{autorización} ocurre cuando el sistema operativo compara la acción solicitada con los permisos del rol.
\end{frame}

\begin{frame}{La autorización reduce acciones indebidas}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\begin{exampleblock}{Ejemplo}
\small Un vendedor solicita \emph{actualizar el stock}. El sistema operativo revisa su rol y permite la acción porque posee ese permiso.
\end{exampleblock}
\pregunta{\small Predice antes de ejecutar:
\begin{enumerate}\setlength\itemsep{2pt}
  \item ¿Qué ocurre si un cliente registra un pedido?
  \item ¿Podrá un cliente modificar el stock?
  \item ¿Y si cualquiera gestionara cuentas?
\end{enumerate}}
\column{.47\textwidth}
\begin{block}{Principio de mínimo privilegio}
\small Cada usuario recibe \textbf{solo} los permisos que necesita para su tarea, y nada más. Así, un error o una cuenta robada causan el menor daño posible.
\end{block}
\vspace{0.15cm}
\begin{exampleblock}{Conclusión del bloque}
\small El sistema operativo utiliza identidades, grupos y permisos para controlar el acceso a recursos.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Antes de seguir: ¿qué es exactamente un «servicio»?}
\concepto{Servicio}{Programa que se ejecuta en segundo plano, sin que nadie lo abra, esperando peticiones para hacer una tarea concreta.}
\begin{center}
\small
\begin{tabular}{>{\bfseries}p{2.7cm}p{5.4cm}p{4.0cm}}
\toprule
& ¿Quién lo inicia? & ¿A quién atiende? \\
\midrule
Programa & Tú, cuando lo abres & A ti, en su ventana \\
Servicio & El sistema operativo, al encender & A su propio equipo \\
Servicio de red & El sistema operativo, al encender & A otros equipos, por la red \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.15cm}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\begin{exampleblock}{Ya tienes servicios funcionando}
\small En tu laptop, ahora mismo: la cola de impresión, las actualizaciones automáticas y la sincronización de la hora. Nunca los abriste y no los verás en una ventana.
\end{exampleblock}
\column{.47\textwidth}
\analogia{Un programa es el chef que llamas cuando quieres cocinar. Un servicio es el cocinero que ya está en la cocina, listo, aunque nadie haya pedido nada todavía.}
\end{columns}
\end{frame}

% 15 ---------------------------------------------------------------------
\begin{frame}{Un servicio de red atiende una necesidad concreta}
\concepto{Servicio de red}{Programa que recibe solicitudes de otros equipos y produce una respuesta.}
\begin{center}
\small
\begin{tabular}{>{\bfseries}p{2.8cm}p{5.0cm}p{3.4cm}}
\toprule
Servicio & Necesidad que atiende & Nombre técnico \\
\midrule
Web & Entrega páginas o aplicaciones & HTTP / HTTPS \\
Nombres & Relaciona nombres con direcciones de red & DNS \\
Configuración automática & Entrega configuración de red a los equipos & DHCP \\
Archivos & Almacena y comparte documentos & SMB / NFS \\
Acceso remoto & Permite administrar un equipo desde otra ubicación & SSH / RDP \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.1cm}
\small Verás estas siglas en cualquier panel de administración, incluida la consola de GCP.
\end{frame}

\begin{frame}{Virtualizar es representar un recurso mediante software}
\concepto{Virtualización}{Creación mediante software de una representación de un recurso tecnológico.}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Recurso físico}
\small Existe en el equipo: procesador, memoria y almacenamiento. Se puede tocar.
\end{block}
\column{.48\textwidth}
\begin{exampleblock}{Representación virtual}
\small Recibe una porción controlada de esos recursos y se comporta como un equipo separado.
\end{exampleblock}
\end{columns}
\vspace{0.15cm}
\analogia{Un edificio dividido en departamentos: cada uno tiene su puerta, su llave y sus muebles, pero todos comparten la misma estructura y el mismo terreno.}
\vspace{0.05cm}
{\small\textbf{\color{utpRojo}Ojo:} virtualizar \textbf{no} crea recursos nuevos; organiza y reparte los que ya existen. Dividir un edificio no agranda el terreno.\par}
\end{frame}

\begin{frame}{Cuatro piezas forman una máquina virtual}
\begin{center}
\small
\begin{tabular}{>{\bfseries}p{3.1cm}p{4.5cm}p{3.6cm}}
\toprule
Pieza & Qué es & En tu laptop sería \\
\midrule
Equipo anfitrión (host) & Computadora física que proporciona sus recursos & Tu laptop \\
Hipervisor & Software que crea y administra máquinas virtuales & \marca{VMware} \\
Máquina virtual (VM) & Computadora hecha con software que recibe parte de los recursos físicos & La ventana que se abre \\
Sistema operativo invitado (guest) & Sistema operativo instalado dentro de la VM & \marca{Lubuntu} dentro de esa ventana \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.15cm}
\begin{alertblock}{Relación clave}
El anfitrión (host) aporta recursos; el hipervisor los asigna; la VM los recibe; el sistema operativo invitado (guest) los utiliza.
\end{alertblock}
\end{frame}

\toprule
Recurso & Para qué sirve & Comparación \\
\midrule
Memoria (RAM) & Guarda lo que se está usando en este momento & El escritorio donde trabajas: si es chico, no cabe todo \\
CPU (núcleos) & Ejecuta las instrucciones & Las manos que hacen el trabajo: más núcleos, más tareas a la vez \\
Almacenamiento (disco) & Conserva los datos aunque se apague & El archivador donde guardas los documentos \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.15cm}
\begin{block}{Diferencia que suele confundir}
\small La memoria se borra al apagar el equipo; el almacenamiento no. Por eso los pedidos se guardan en disco y no en memoria.
\end{block}
\end{frame}

% 23 ---------------------------------------------------------------------
\begin{frame}{¿Por qué separar en dos VM y no instalar todo junto?}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Todo en un mismo sistema operativo}
\small
\begin{itemize}\setlength\itemsep{2pt}
  \item Una falla afecta a todo el sitio.
  \item Actualizar la tienda obliga a tocar los archivos.
  \item Quien entra a la web alcanza los archivos.
  \item Los programas se pelean por la memoria.
\end{itemize}
\end{block}
\column{.48\textwidth}
\begin{exampleblock}{Separado en dos VM}
\small
\begin{itemize}\setlength\itemsep{2pt}
  \item Si una VM falla, la otra sigue.
  \item Cada VM se actualiza por su cuenta.
  \item Los permisos se definen por VM.
  \item Cada VM tiene un límite de recursos.
\end{itemize}
\end{exampleblock}
\end{columns}
\vspace{0.2cm}
\ojo{Separar también cuesta: cada VM necesita su propio sistema operativo, su memoria y su mantenimiento. Se separa cuando el beneficio supera ese costo, no por costumbre.}
\end{frame}

\begin{center}

\end{center}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\begin{block}{Aislamiento}
\small Cada VM es un equipo independiente: su sistema operativo, su disco y su IP.
\end{block}
\column{.47\textwidth}
\begin{exampleblock}{En la práctica}
\small Reinicias Kali Linux y Lubuntu sigue funcionando. Pero si apagas PC-JUAN, se apagan las dos.
\end{exampleblock}
\end{columns}
\end{frame}

\begin{frame}{Asignar a una VM significa reservar capacidad física}
\begin{center}
\small
\begin{tabular}{>{\bfseries}p{3.6cm}ccc}
\toprule
Anfitrión (host) disponible & Memoria & CPU & Almacenamiento \\
\midrule
PC-JUAN & 8 GB & 4 núcleos & 100 GB \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.25cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Asignado}
\small Suma de los recursos entregados a Kali Linux y Lubuntu. Quedan reservados aunque la VM esté ociosa.
\end{block}
\column{.48\textwidth}
\begin{block}{Sin asignar}
\small Capacidad restante que el hipervisor todavía puede entregar a otra VM.
\end{block}
\end{columns}
\vspace{0.2cm}
\centering\large Sin asignar = Capacidad total $-$ Total asignado
\end{frame}

% 26 ---------------------------------------------------------------------
\begin{frame}{Una configuración válida respeta los tres límites}
\begin{center}
\small
\begin{tabular}{>{\bfseries}p{3.1cm}ccc}
\toprule
Máquina virtual & Memoria & CPU & Almacenamiento \\
\midrule
Kali Linux & 3 GB & 2 núcleos & 35 GB \\
Lubuntu & 3 GB & 1 núcleo & 45 GB \\
\midrule
Total asignado & 6 GB & 3 núcleos & 80 GB \\
Sin asignar & 2 GB & 1 núcleo & 20 GB \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.15cm}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\begin{exampleblock}{La cuenta, recurso por recurso}
\small Memoria: $3+3=6 \leq 8$ GB\\
CPU: $2+1=3 \leq 4$ núcleos\\
Almacenamiento: $35+45=80 \leq 100$ GB
\end{exampleblock}
\column{.47\textwidth}
\begin{alertblock}{Basta con que uno falle}
\small Si el almacenamiento sumara 120 GB, la configuración se rechaza aunque memoria y CPU estén bien.
\end{alertblock}
\end{columns}
\end{frame}
